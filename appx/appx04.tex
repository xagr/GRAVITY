\chapter{流形及其张量场}\label{appx:manifold}

最初仅用微积分和坐标系研究“背景”的\textbf{局部}(local)性质，建立了\textbf{微分几何}(differential geometry)。随着集合论和拓扑学发展，我们发现可将“背景”准确定义为拓扑空间，且拓扑学提供了分析\textbf{整体}(global)性质的工具，便有了\textbf{现代微分几何}。

\section{微分流形}
物理背景作为拓扑空间，是无特征且无定形的，其中的点和曲线是不能先验识别或定义的，除非利用坐标系。有用的背景从来不可能真正脱离坐标。
坐标方法就是一种给背景点连续指定 $\F^n$ 数据的映射。
例如，平面上附着不同的坐标空间，一种可存放正交坐标，一种可存放极坐标，从而可赋予某点 $p$ 正交坐标 $(0,1)$，也可赋予极坐标 $(1,\pi/2)$，如图 \ref{fig:coor-chan}。拓扑学的同胚关系可描述坐标的指定，因为同胚具有双向连续性，且两个点不会被分配到同一坐标，从而避免歧义。
当然，同胚使得许多坐标系能力有限，不能揽下全局。比如，极坐标系不能完整覆盖 $\R^2$，坐标取值为 $V=(0,+\infty)\times(0,2\pi)$。因此常考察开子集。为圆满完成指派任务，整个背景须由若干坐标系拼接，且互相分担对方力所不及之地。本节主要考虑 $\R^n$。


\begin{definition}
    置 $(M,\topo T)$。存在 $n\in\N$，使 $M$ 任意点都存在邻域同胚于 $(\R^n,\topo T_u)$ 某开集，则称其\textbf{局部欧}。
    $\psi$ 称为\textbf{坐标映射}，$\psi(p)$ 称为 $p$ 的\textbf{坐标}。$O$ 称为\textbf{坐标域}(patch)，$(O,\psi)$ 称为\textbf{坐标系}或\textbf{图}(chart)。相应的坐标系族 $\mathcal A=\{(O,\psi):O\in\topo P\}$ 称为\textbf{图册}(atlas)。$(M,\mathcal A)$ 称为\textbf{局部欧氏空间}。维数为 $\dim M:=n$。当 $\mathcal A$ 自明时简记 $M$。
\end{definition}

\begin{remark}
    同胚连续性由全集拓扑或诱导拓扑衡量。原则上可修改定义，使其允许各坐标系不同维，这样空间无唯一维数，但实用价值不大。
\end{remark}

\begin{theorem}
    局部欧的等价说法是任意点存在邻域同胚于开球或 $\R^n$。
\end{theorem}
\begin{proof}
    因为 $\psi(x)$ 的邻域内一定存在包含该点的开球。再注意开球与 $\R^n$ 同胚。
\end{proof}

同胚赋予的第 $i$ 个坐标构成映射 $x^i:O\to\R$，坐标系又常记 $\{x\}$。
就某个坐标系而言，坐标域的子集
    $\{p\in O: x^i(p)$ 是常函数，$i=2,\cdots,n\}$
可看成以 $x^1$ 为参数的路径，称其是一条 $x^1$ \textbf{坐标线}(coordinate line)。其它情况以此类推。全体坐标线构成\textbf{坐标网格}。
坐标域是某点邻域，故坐标系又称局部坐标系，同胚又称局部同胚。仅用一个坐标域就能覆盖则称\textbf{平凡的}(trivial)，此时可称坐标系是整体坐标系。典型的例子是 $\R^n$ 自身。设 $a=(x^1_a,\cdots,x^n_a)\in\R^n$，恒等映射给出\textbf{自然坐标} $\psi(a)=(x^1(a),\cdots,x^n(a))=a$。

\begin{definition}
    仿紧 $T_2$ 局部欧氏空间称为\textbf{拓扑流形}，简记 $M$。将 $\R^n$ 换为 $\C^n$ 就是\textbf{复流形}。
\end{definition}

\begin{remark}
    仿紧 $T_2$ 是为了应用单位分解。
    也可用更严苛的 $C_2,T_2$ 条件。
    下一节将说明 $C_2\land T_2\land\text{局部欧}\im \text{仿紧}\land T_2\land\text{局部欧} \im \text{可度量}$。
\end{remark}

由于坐标域都是开的，故必存在相交之地。此处便涉及坐标系的转化。

\begin{definition}
    取 $(O_\alpha,\psi_\alpha),(O_\beta,\psi_\beta)\in \mathcal A$。若 $O_\alpha\cap O_\beta=\mt$；或 $O_\alpha\cap O_\beta\ne\mt$ 时\textbf{坐标变换}或\textbf{转移函数} $\psi_\alpha\circ\psi_\beta^{-1},\psi_\beta\circ\psi_\alpha^{-1}$ 的 $r$ 阶导函数连续（记 $C^r$），则称两坐标系 $C^r$\textbf{-相容}。
    所含坐标系皆 $C^r$-相容的 $\mathcal A$ 称为 $C^r$\textbf{-图册}。$C^\infty$-相容简称\textbf{相容}。
\end{definition}

\begin{remark}
    坐标变换可显式地表为 $n$ 个 $n$ 元函数，$C^r$ 性由 $\R^n,\C^n$ 分析学给出。$C^0$ 即连续。光滑即任意阶导函数连续，记 $C^\infty$。解析是指任意点可展开为收敛于自身的幂级数，从而一定光滑。由于光滑仍可做到在某点无穷阶平坦，如 $y=\e^{-1/x}$ 于 $x=0$（左侧 $x\leqslant 0$ 连接 $y=0$），因而从该点展开收敛不到自身，即光滑不一定解析。由此可构造实用的\textbf{隆起函数}(bump function)
    \eq{
        \Psi(x)=\begin{cases}
\exp(\frac{1}{x^2-1}), & |x|<1,\\
0, & |x|\geqslant 1.
\end{cases}
    }
    考虑复变函数 $f(z)=u(z)+\i v(z)$ 对定义域内 $z=x+\i y$ 可导，即 $f'(z):=\lim_{\Delta z\to 0}(f(z+\Delta z)-f(z))/\Delta z$ 需要对任意 $\Delta z=\Delta x+\i\Delta y$ 收敛。只需分别取 $\Delta x,\Delta y=0$ 就有 $\pdv{u}{x}+\i\pdv{v}{x}=\frac{1}{\i}\left(\pdv{u}{y}+\i\pdv{v}{y}\right)$，即 Cauchy-Riemann 条件
    \eq{
        \pdv{u}{x} = \pdv{v}{y},\quad \pdv{v}{x} = -\pdv{u}{y}.
    }
    因此虽然 $\C,\R^2$ 同胚，但 $\C$ 多出来乘法、线性等结构使复可导要比实可导多出上述条件，从而可证复变函数只需可导（称为全纯）就等价于解析。
\end{remark}

\begin{example}
    拓扑流形相当于仅要求 $C^0$-相容。
\end{example}

假设有两个图册 $\mathcal A,\mathcal B$。它们可能并不相容，即其间存在相交但不相容的坐标系。如果它们相容，则可合并为一个更大的图册。当然希望将所有相容的坐标系都囊括进来，尽可能构成足够大的图册。

\begin{definition}
    一个不真含于任何 $C^r$-图册的 $C^r$-图册称为\textbf{最大的}(maximal)。若局部欧氏空间的图册是 $C^r$-最大图册，则称 \textbf{$C^r$-流形}。最大图册又称\textbf{微分结构}。$C^\infty$-流形称为\textbf{微分流形}或\textbf{光滑流形}。
\end{definition}

\begin{example}
    \textbf{通常流形}就是指 $(\R^n,\topo T_u)$ 赋以包含自然坐标系的最大光滑图册。
\end{example}

综上，将今后常用的 $C^\infty$-拓扑流形之定义总结为：
\begin{tcolorbox}
\begin{definition}
    置 $T_2$ 仿紧空间 $(M,\topo T)$，若 $\exists\topo P\subset\topo T$ 使
    \begin{itemize}
        \item $\topo P$ 是开覆盖：$M = \bigcup_{O\in\topo P}O$；
        \item $\forall O\in\topo P$，$\exists$ 同胚 $\psi:O\to V\in\topo T_u$（或开球、$\R^n$）；
        \item $O_\alpha\cap O_\beta\ne\mt$ 时 $\psi_\alpha\circ\psi_\beta^{-1}$ 光滑，
    \end{itemize}
    则称 $\{(O,\psi):O\in\topo P\}$ 为光滑图册。若一个光滑图册 $\mathcal A$ 不真含于任何光滑图册，则称 $(M,\topo T,\mathcal A)$ 为一个\textbf{光滑拓扑流形}，简称\textbf{流形}。
\end{definition}
\end{tcolorbox}
或简言之，
\begin{center}
    \tcbox{
    \textbf{流形}是拓扑性质良好的空间，赋有相互光滑变换的一切局部坐标系。
    }
\end{center}

定量计算流形间的映射，必然使用坐标系，即通过 $\psi'\circ f\circ \psi^{-1}$ 来确定 $f$。
\begin{definition}
    置 $C^k$-流形 $M,M'$。$f:M\to M'$ 称为 $C^r$\textbf{-映射}，若 $\forall p\in M$，存在坐标系 $(\psi,O),(\psi',O')$ 使 $p\in O,f(p)\in O'$ 且 $\psi'\circ f\circ \psi^{-1}$ 是 $C^r$ 的（$r\leqslant k$）。
\end{definition}

\begin{remark}
    $r\leqslant k$ 使 $C^r$ 性与坐标选择无关。此后类似概念的良定义性均同理。
\end{remark}

\begin{example}
    坐标变换 $\psi'\circ \psi^{-1}$ 相当于取恒等 $f=\id_M:M\to M$。
\end{example}
\begin{example}
    曲线是流形间 $C^0$-映射 $C:I\to M$ 的特例。其中将 $I\subset\R$ 上坐标系视作恒等映射，从而只需考察\textbf{参数方程} $\psi\circ C$ 的性质。
\end{example}

\begin{definition}
    流形上的曲线为 $C^r$-映射就称 $C^r$\textbf{-曲线}。有限个光滑曲线收尾相连构成\textbf{分段光滑曲线}。
\end{definition}

\begin{definition}
    \textbf{$C^r$-同胚}是流形间双向 $C^r$ 的双射。$C^\infty$-同胚称为\textbf{微分同胚}或\textbf{光滑同胚}。
\end{definition}

\begin{theorem}
    开球和 $\R^n$ 微分同胚
\end{theorem}
\begin{proof}
    不失一般性，$B(0,1),\R^n$ 的同胚式 \eqref{eq:B-Rn} 是双向光滑的。
\end{proof}

微分同胚的流形可建立起等价关系，故我们常考虑的微分同胚实际上是自同胚。
看待微分同胚 $f:M\to M$ 有两种不同观点：将 $f$ 用同一个坐标系表为 $\psi\circ f\circ \psi^{-1}$，此即“点动、坐标不动”的主动观点；定义 $\psi':=\psi\circ f$ 就有坐标变换 $\psi' \circ\psi^{-1}$，此即“点不动、坐标动”的被动观点。

以（光滑拓扑）流形为对象、光滑映射为态射，构成\textbf{流形范畴} $\cate{Man}$，同构态射即微分同胚。


\begin{example}
    流形上的函数是流形间映射 $f:M\to\F$ 的特例。其中将 $\F=\R,\C$ 上坐标系视作恒等映射，从而只需考察 $n$ 元函数 $f\circ\psi^{-1}$ 的性质。
\end{example}

\begin{definition}
    流形上的函数称为\textbf{标量场}(scalar field)。$U\subset M$ 上全体 $C^r$-标量场构成的集合记作 $C^r(U)$。
\end{definition}

我们曾说明 $\F^X$ 是 $\F$-模。原则上还可引入乘法：$\forall p\in X,f,g\in \F^X$，
\eq{(f+g)(p):=f(p)+g(p),\quad  (fg)(p):=f(p)g(p),}
将常函数的赋值 $\alpha\in\F$ 就记为函数名，即 $\alpha(p):=\alpha$，于是数乘即用常函数相乘，同时也就囊括了零元。故 $\F^X$ 同时是 $\F$-模和交换幺环，称\textbf{函数环}。且注意可有 $\alpha(fg)=(\alpha f) g=f(\alpha g)$，从而 $\F^X$ 还是 $\F$-代数。
但 $\F^X$ 不能是域，因为非零元不一定有逆：设 $f\ne 0$ 但存在零点，则不存在 $g$ 使 $f(p)g(p)=1$ 恒成立。

现在考虑 $C^r(U)$。$C^r$-性保持了运算封闭，故 $C^r(U)$ 仍是交换幺 $\F$-代数，可称 $U$ 上的 $C^r$-函数环。为方便，我们常考虑实光滑函数。

\begin{example}
    $C^k$-流形上的第 $i$ \textbf{坐标映射} $x^i:O\to\R$ 属于 $C^k(O)$。
\end{example}

\section{流形的拓扑性质*}

\begin{theorem}
    局部欧 $\im$ 局部路径连通 $\land$ 局部可度量；若再满足 $T_2$，则局部紧。
\end{theorem}
\begin{proof}
    任给 $p\in M$ 及其邻域 $N$，取 $(O,\psi)$ 使 $p\in O$。存在 $\psi(p)$ 为心的欧氏开球 $B\subset \psi[O\cap N^\circ]$，同胚保持路径连通性，则 $\psi^{-1}[B]\subset N$ 是 $p$ 的路径连通开邻域，故 $M$ 局部路径连通。

    坐标域 $O$ 上的度量 $d(p,q):=\|\psi(p)-\psi(q)\|$ 诱导其子拓扑，故 $M$ 局部可度量。

    若 $M$ 为 $T_2$，取上述 $B$ 使闭球 $\bar B\subset\psi[O]$。则 $K:=\psi^{-1}[\bar B]$ 紧，因而在 $M$ 中闭。于是开邻域 $U:=\psi^{-1}[B]$ 在 $M$ 中的闭包满足 $\bar U\subset K$，故紧，即 $M$ 局部紧。
\end{proof}

\begin{theorem}
    路径连通 $\xLeftrightarrow{\text{局部欧}}$ 连通。
\end{theorem}
\begin{proof}
    已知路径连通 $\im$ 连通。反之，局部欧 $\land$ 连通 $\im$ 局部路径连通 $\land$ 连通 $\im$ 路径连通。
\end{proof}

\begin{theorem}
    拓扑流形可度量。
\end{theorem}
\begin{proof}
    定义一：$\text{仿紧} \land T_2 \land \text{局部欧}\im \text{仿紧} \land T_2 \land \text{局部可度量} \xRightarrow{\text{Smirnov}}$ 可度量。

    定义二：$C_2 \land T_2 \land \text{局部欧} \im C_2 \land T_2 \land \text{局部紧} \im C_2 \land T_3 \xRightarrow{\text{Urysohn}}$ 可度量。
\end{proof}
\begin{remark}
    注意可度量 $\im$ 仿紧，所以其实定义二 $\im$ 定义一。
\end{remark}

\begin{definition}
    若开覆盖 $\topo P=\{O_i\}$ 的单位分解 $\{\rho_i\}$ 光滑，则称\textbf{光滑单位分解}。
\end{definition}

\begin{lemma}[光滑 Urysohn 引理]\label{thm:smooth_urysohn_bump}
    设 $M$ 为光滑流形，$K\subset U\subset M$，其中 $K$ 紧、$U$ 开。则存在 $\varphi\in C^\infty(M)$，使 $0\leqslant\varphi\leqslant1$、$\varphi|_K=1$，且 $\supp\varphi$ 为含于 $U$ 的紧集。
\end{lemma}
\begin{proof}
    置 $h(t)=\e^{-1/t}$（$t>0$），$h(t)=0$（$t\leqslant0$）。由前文知 $h$ 光滑，故
    \[
        \chi(t):=\frac{h(1-t)}{h(1-t)+h(t-1/4)}
    \]
    光滑，取值于 $[0,1]$，且在 $t\leqslant1/4$ 时为 $1$，在 $t\geqslant1$ 时为 $0$。

    对每个 $x\in K$，取坐标系 $(O_x,\psi_x)$ 使 $x\in O_x\subset U$、$\psi_x(x)=0$，并取 $r_x>0$ 使 $\bar B(0,r_x)\subset\psi_x[O_x]$。在 $O_x$ 上置
    \[
        b_x(p):=\chi\!\left(\frac{\|\psi_x(p)\|^2}{r_x^2}\right),
    \]
    并在 $M\backslash O_x$ 上置零。其支撑含于紧集 $\psi_x^{-1}[\bar B(0,r_x)]\subset O_x$，故延拓后仍光滑；且 $b_x$ 在开邻域 $V_x:=\psi_x^{-1}[B(0,r_x/2)]$ 上恒为 $1$。

    由紧性，取 $V_{x_1},\ldots,V_{x_N}$ 覆盖 $K$，令
    \[
        \varphi:=1-\prod_{i=1}^N(1-b_{x_i}).
    \]
    则 $\varphi$ 光滑，$0\leqslant\varphi\leqslant1$，在 $K$ 上恒为 $1$，且 $\supp\varphi\subset\bigcup_{i=1}^N\supp b_{x_i}\subset U$，故支撑紧。$K=\mt$ 时取 $\varphi=0$ 即可。
\end{proof}

\begin{theorem}
    光滑流形的任何开覆盖均存在从属于它的局部有限的光滑单位分解。
\end{theorem}
\begin{proof}
    设开覆盖 $\{O_\alpha\}$。由局部紧性，先取相对紧开集组成的加细，再由仿紧性取其局部有限开加细 $\{U_i\}$，其中 $U_i\subset O_{\alpha(i)}$、$\bar U_i$ 紧。沿用连续单位分解证明中的收缩构造，取闭覆盖 $\{F_i\}$ 使 $F_i\subset U_i$，则各 $F_i$ 紧。

    由引理 \ref{thm:smooth_urysohn_bump}，取光滑函数 $b_i$ 使 $0\leqslant b_i\leqslant1$、$b_i|_{F_i}=1$、$\supp b_i\subset U_i$。令
    \[
        b:=\sum_i b_i,\quad
        \rho_\alpha:=\frac{\sum_{\alpha(i)=\alpha}b_i}{b}.
    \]
    局部有限性保证各和光滑，且 $\{F_i\}$ 覆盖 $M$ 保证 $b\geqslant1$。故 $\rho_\alpha$ 光滑，$0\leqslant\rho_\alpha\leqslant1$，$\sum_\alpha\rho_\alpha=1$。又由局部有限性，$\bigcup_{\alpha(i)=\alpha}\supp b_i$ 闭，故
    \[
        \supp\rho_\alpha\subset\bigcup_{\alpha(i)=\alpha}\supp b_i\subset O_\alpha.
    \]
    每点附近只遇到有限个 $U_i$，故也只遇到有限个 $\supp\rho_\alpha$，即得所求。
\end{proof}


\section{切空间}\label{chdm:sec_tangentspace}
\label{chdm:sec_tangent-bundles}\label{chdm:sec_tangent-vector-field}
\label{chdm:sec_tensor-fields}\label{chdm:sec_vdef2}\label{chdm:sec_poisson-lie}

物理学的矢量总是靠作差或求导来产生，即切矢。曾指出给定某几何形体的参数方程 $x^\mu(t,\cdots)$ 后，可得到切矢场 $T^\mu = \pdv*{x^\mu}{t}$，它具有变换 $T'^\nu = \pdv{x'^\nu}{x^\mu}T^\mu$。而其中 $\pdv*{x'^\nu}{x^\mu}$ 就是 $\{x\}$ 的坐标基，故这相当于将切矢展开到 $\{x\}$ 坐标基 $T'^\nu = T^\mu \pdv{x'^\nu}{x^\mu}$，但最终用 $\{x'\}$ 表达。
可见，坐标分量 $T'^\nu$ 是 $\pdv{t}=\pdv{x^\mu}{t}\pdv{x^\mu}=T^\mu \p_\mu$ 对 $x'^\nu:O\to\R$ 作用的结果，故仅需偏导数本身配合所作用的 $x'^\nu(x)$ 即可给出我们对切矢要求的所有性质。
立即发现，这种描述已独立于坐标语言。数学又常常使用映射来构造概念，故何不直接定义 $T=\pdv*{t}$？
Cartan 最先提出这一思想：用一个具有求导性质的泛函来代指切矢。

将切矢定义为一个求导映射，与最传统的曲线相切有何直观联系？
给定一条穿过 $p\in M$ 的曲线 $\gamma$ 且在 $p=\gamma(0)$ 处可导，规定给出相同切矢的曲线等价，相同性通过坐标分量 $\dv{t}\big|_{t=0}(\psi\circ\gamma)$ 来判断。
置任意可导标量场 $f$，若某点切矢为 $A^\mu$，则方向导数为 $\dv*{f}{t}=A^\mu\p_\mu f$。故等价类 $[\gamma]$ 中的曲线给出了相同的方向导数 $\dv*{t}$，即曲线等价类与方向导数一一对应。
标量场在此处的作用是引出切矢。犹如在电场中放入测试电荷，用静电力研究电场一样，标量场受导数作用出实数，以研究导数自身，而标量场不是关键。

剩下的问题在于需要哪些求导性质。若直接使用偏导符号 $A=A^\mu\pdv{x^\mu}$，实际上仍然使用了坐标语言，因此要将目光转向导数的代数性质。
若 $f$ 是解析的，展开为
\[
    f=f(p)+\p_i f(p)(x^i-x^i(p))+ \frac 12\p_i\p_j f(p)(x^i-x^i(p))(x^j-x^j(p))+\cdots.
\]
$A=A^\mu\p_\mu$ 是求导操作，至少要考虑到线性性、Leibniz 乘法法则 $A(fg)=g(p)A(f)+f(p)A(g)$。
乘法法则给出 $A(1)=A(1\cdot 1)=2A(1)\Rightarrow A(1)=0$，再由线性性知$\forall c\in\R$，$A(c)=A(c\cdot 1)=c A(1)=0$。
从而 $A$ 作用于 $f$ 的确只剩下 $A(f)=A(x^\mu)\p_\mu f(p)$，高阶项为零是因为必然存在 $x^j(p)-x^j(p)=0$。可见线性性、乘法法则已能反求得 $A=A^\mu\p_\mu$，故此即所需的全部求导代数性质了。
当然，对无穷级数滥用代数运算是危险的，其次也不需要 $f$ 解析，我们更倾向于考虑有限项相加。

需要注意，坐标函数 $x^\mu$ 通常只在坐标域上定义，好在求导只关心 $p$ 附近的函数值：只要 $M$ 上函数 $f_1,f_2$ 在 $U\in\topo N(p)$ 上相等，置 $f:=f_1-f_2$，令 $h$ 在 $M\backslash U$ 上为零但 $h(p)\ne 0$，就可构造零元 $fh=0$，则 $0=A(fh)=h(p)A(f)$ 从而 $A(f)=0$，即 $p$ 点任意切矢 $A$ 满足 $A(f_1)=A(f_2)$。便能合法谈及坐标分量 $A(x^\mu)$。

\begin{definition}
    置光滑流形 $M$ 及 $p\in M$。设 $U,V$ 为 $p$ 的开邻域，$f\in C^\infty(U),g\in C^\infty(V)$ 为实函数。若存在开邻域 $W\ni p$，使 $W\subset U\cap V$ 且 $f|_W=g|_W$，则称 $f,g$ 在 $p$ 处等价。等价类 $[f]_p$ 称为 $p$ 处的\textbf{函数芽}，全体记作 $C^\infty_p(M)$。在共同定义域上作运算，规定
    \eq{
        [f]_p+[g]_p:=[f+g]_p,\quad c[f]_p:=[cf]_p,\quad [f]_p[g]_p:=[fg]_p,
    }
    则 $C^\infty_p(M)$ 构成实代数，且 $[f]_p(p):=f(p)$ 与代表元无关。

    $p$ 处的\textbf{切矢}是满足 Leibniz 乘法法则的实线性映射 $v:C^\infty_p(M)\to\R$，即
    \eq{
        v([f]_p[g]_p)=g(p)v([f]_p)+f(p)v([g]_p).
    }
    对任何在 $p$ 附近定义的光滑函数，总可简记 $v([f]_p)=v(f)$。
    $p$ 处全体切矢的集合 $T_p M$ 称为 $M$ 在 $p$ 处的\textbf{切空间}。全体切空间的无交并
    \eq{\label{chdm:eqn_tangent-bundle-set}
        TM:=\bigsqcup_{p\in M} T_p M=\bigcup_{p\in M}\{p\}\times T_p M=\{(p,v):p\in M,v\in T_p M\}
    }
    称为\textbf{切丛}(tangent bundle)\footnote{$M$ 上实函数 $f(x)$ 的像 $(x,f(x))$ 便是从 $M$ 到卡氏积流形 $M\times \R$ 的一个映射；这个概念可以进一步推广成纤维丛，本节的切丛是其简单例子。}。
\end{definition}

\begin{remark}
    直观图像是把切空间理解为背景在该点的切平面，切丛就是全体切平面。

    此处直接考虑光滑性 $r=\infty$，是为了使坐标分解
    \[
        f-f(p)=\sum_i(x^i-x^i(p))H_i
    \]
    中的 $H_i$ 仍属于同一函数代数，从而可对各项使用 Leibniz 法则。对 $f\in C^r$（$1\leqslant r<\infty$），这种分解的系数一般仅为 $C^{r-1}$，$v(H_i)$ 未必有定义；光滑时则无此问题，并可由此证明切矢恰由 $\dim M$ 个坐标分量决定。因此不能直接将上述定义中的 $\infty$ 换为有限 $r$；有限阶可微流形可用曲线等价类定义切矢。
\end{remark}

\begin{theorem}
    前文已暗示更简单的等价定义：$p$ 处的切矢是满足
    \[
        D(fg)=g(p)D(f)+f(p)D(g)
    \]
    的实线性泛函 $D:C^\infty(M)\to\R$。两种定义的对应关系为 $D(f)=v([f]_p)$。
\end{theorem}
\begin{proof}
    由函数芽上的切矢 $v$ 定义 $D(f):=v([f]_p)$，显然满足上述条件。

    反之，先证 $D$ 只依赖函数芽。若 $h\in C^\infty(M)$ 在 $p$ 的开邻域 $W$ 上恒为零，取光滑函数 $\chi$ 使 $\chi(p)=1$、$\supp\chi\subset W$，则 $h\chi=0$，故
    \[
        0=D(h\chi)=\chi(p)D(h)+h(p)D(\chi)=D(h).
    \]
    因而在 $p$ 附近相等的全局函数具有相同的 $D$ 值。

    任给函数芽 $[f]_p$，设 $f\in C^\infty(U)$。取光滑函数 $\rho$，使其在 $p$ 的某邻域上恒为 $1$，且 $\supp\rho\subset U$。将 $\rho f$ 在 $M\backslash U$ 上置零，得到 $F\in C^\infty(M)$，且 $[F]_p=[f]_p$。于是 $v([f]_p):=D(F)$ 与代表元及延拓的选择无关，并保持线性性与 Leibniz 法则。每个芽都有这样的全局代表，故此对应唯一且互逆。
\end{proof}

\begin{theorem}
    $T_p M$ 是线性空间，且 $\dim T_p M=\dim M =: n$。
\end{theorem}
\begin{proof}
    切矢对通常的泛函加法及数乘封闭，故 $T_p M$ 为实线性空间。取坐标系 $(O,\psi)$，置 $a:=\psi(p)$。由求导法则，
    \[
        \p_i|_p(f):=\pdv{(f\circ\psi^{-1})}{x^i}\bigg|_a
    \]
    是切矢，且 $\p_i|_p(x^j)=\delta_i^j$，故这 $n$ 个切矢线性无关。此即 $\dim T_p M\geqslant n$。

    任给 $f$ 在 $p$ 处的光滑函数芽，缩小 $O$，使 $f$ 在 $O$ 上光滑且 $\psi[O]$ 为以 $a$ 为心的开球。记 $\hat f:=f\circ\psi^{-1}$，由微积分基本定理（对 $i$ 用求和约定），
    \[
        f(q)-f(p)= (x^i(q)-x^i(p))H_i(q),\quad
        H_i(q):=\int_0^1\d{t}\p_i\hat f\bigl(a+t(\psi(q)-a)\bigr).
    \]
    各 $H_i$ 光滑，且 $H_i(p)=\p_i|_p(f)$。任给 $v\in T_p M$，由 $v(c)=0$ 及 Leibniz 法则，得
    \[
        v(f)= v(x^i)H_i(p)= v(x^i)\p_i|_p(f).
    \]
    因 $f$ 任意，故
    \eq{\label{chdm:eqn_v-expand-on-natural-bases}
        v= v^i\p_i|_p,\quad v^i:=v(x^i).
    }
    于是 $\{\p_i|_p\}_{i=1}^n$ 张成 $T_p M$，从而是一组基，故 $\dim T_p M=n$。
\end{proof}

\begin{definition}
    $X_i = \p_i|_p$ 称为 $p$ 处第 $i$ 个\textbf{坐标基矢}。$\{X_i\}$ 称为 $p$ 处的\textbf{坐标基}。
\end{definition}

\begin{theorem}[坐标分量变换律]
    置坐标系 $(O,\psi),(O',\psi')$ 及 $p\in O\cap O'$。两坐标基及同一切矢 $v\in T_pM$ 的分量满足
    \eq{\label{chdm:eqn_xy-transform}
        X_i= {\pdv{x'^j}{x^i}}(p)X'_j,\quad
        v'^j= {\pdv{x'^j}{x^i}}(p)v^i, \quad {\pdv{x'^j}{x^i}}(p):=\pdv{(x'^j\circ\psi^{-1})}{x^i}\bigg|_{\psi(p)}.
    }
    其中 Jacobi 系数 $\pdv*{x'^j}{x^i}$ 是 $O\cap O'$ 上的光滑函数。
\end{theorem}
\begin{proof}
    任给 $p$ 处的光滑函数芽 $f$，由 $f\circ\psi^{-1}=(f\circ\psi'^{-1})\circ(\psi'\circ\psi^{-1})$ 及链式法则，得
    \[
        X_i(f)= {\pdv{x'^j}{x^i}}(p)X'_j(f).
    \]
    因 $f$ 任意，即得坐标基变换。将其代入 $v= v^iX_i= v'^jX'_j$，比较系数即得分量变换。
\end{proof}

切丛不仅是切空间的集合，还自然带有光滑结构。令 $\pi:TM\to M$、$\pi(p,v)=p$，称为\textbf{投影}；其\textbf{纤维} $\pi^{-1}[\{p\}]$ 就是 $T_pM$。下文常将 $(p,v)$ 简记为 $v$，所在点由 $v\in T_pM$ 确定。

\begin{theorem}
    $n$ 维光滑流形 $M$ 的切丛 $TM$ 自然为 $2n$ 维光滑流形。在坐标系 $(U,x)$ 上，其坐标映射为
    \eq{\label{chdm:eqn_tb_atalas}
        \widetilde x:\pi^{-1}[U]\to x[U]\times\R^n,\qquad
        (p,v^i\p_i|_p)\longmapsto(x(p),v^1,\ldots,v^n).
    }
    若 $M$ 第二可数，则 $TM$ 也第二可数。
\end{theorem}
\begin{proof}
    各 $\widetilde x$ 均为双射。由切矢分量变换律，若 $h=x'\circ x^{-1}$，则交叠处的坐标变换为
    \[
        (u,v)\longmapsto\bigl(h(u),Dh(u)v\bigr),
    \]
    是欧氏开集间的微分同胚。因此这些坐标中的开集共同给出 $TM$ 的拓扑，且各坐标光滑相容。不同纤维上的两点可用底空间中的不交开邻域分开；同一纤维上的不同点可在同一坐标域中分开，故 $TM$ 为 $T_2$。

    再证仿紧性。由 $M$ 局部紧且仿紧，取局部有限开覆盖 $\{U_\alpha\}$，使各 $\bar U_\alpha$ 紧且含于某坐标域。用从属于坐标覆盖的单位分解，将各纤维上的欧氏范数平方加权求和后开平方，得到连续纤维范数 $r$。开集族
    \[
        W_{\alpha k}:=\pi^{-1}[U_\alpha]\cap\{k-1<r<k+1\},\qquad k=0,1,\ldots,
    \]
    局部有限并覆盖 $TM$，且各闭包紧：在紧底集 $\bar U_\alpha$ 上，$r$ 与坐标中的欧氏范数一致等价，故 $r\leqslant k+1$ 给出有界纤维坐标。对 $TM$ 的任意开覆盖，在每个 $\bar W_{\alpha k}$ 上取有限子覆盖，再与 $W_{\alpha k}$ 相交，即得局部有限开加细。

    若 $M$ 第二可数，由 Lindel\"of 性可选可数个坐标域覆盖 $M$，相应的切丛坐标域也可数；在各坐标像中取可数基，合并便得 $TM$ 的可数基。
\end{proof}

切矢场就是给每点指定该点的一个切矢：
\begin{definition}\label{chdm:def_tangent-vector-field}
    置开集 $U\subset M$。若映射 $v:U\to TM$ 满足 $v(p)\in T_pM$，即 $\pi(v(p))=p$，则称其为 $U$ 上的\textbf{切矢场}，亦称 $TM$ 在 $U$ 上的\textbf{截面}(section)。

    若 $v:U\to TM$ 为 $C^r$ 映射，则称其为 $C^r$ 切矢场。由切丛坐标，这等价于各坐标域中的分量 $v^i$ 均为 $C^r$ 函数。全体光滑切矢场记作 $\Gamma(TM|_U)$，亦记 $\mathfrak X(U)$；$U=M$ 时即 $\mathfrak X(M)=\Gamma(TM)$。本书不标可微阶数的 $\Gamma$ 均指光滑截面的集合。
\end{definition}

在坐标域 $U$ 上，$\{\p_i\}$ 逐点为切空间的基，称为\textbf{坐标标架场}，并有
\eq{\label{chdm:eqn_vfnf}
    v|_U=v^i\p_i.
}
这些坐标基场一般只在 $U$ 上定义。逐点规定加法、实数乘法及函数乘法：
\eq{\label{chdm:eqn_rnumtimesv}
    (u+v)_p=u_p+v_p,\qquad (fv)_p=f(p)v_p,\quad f\in C^\infty(M).
}
常函数给出实数乘法，故 $\mathfrak X(M)$ 既是实线性空间，也是 $C^\infty(M)$-模。在坐标域上，此模以 $\{\p_i\}$ 为基；整体上未必存在这样的标架。若 $M\ne\mt$ 且 $n>0$，则它作为实线性空间为无穷维：在一坐标域内取非零光滑函数 $\chi$，支撑紧且含于该域，将 $\chi(x^1)^k\p_1$（$k=0,1,\ldots$）在域外置零，便得无穷多个实线性无关的光滑场。

光滑切矢场还可看作光滑函数代数上的求导算子：
\eq{\label{chdm:eqn_vecffunc}
    v:C^\infty(M)\to C^\infty(M),\qquad v(f)(p):=v_p(f).
}

\begin{theorem}\label{chdm:thm_vf-def2}
    光滑切矢场与 $C^\infty(M)$ 上的实线性导子一一对应，即与满足 $D(fg)=gD(f)+fD(g)$ 的实线性映射 $D:C^\infty(M)\to C^\infty(M)$ 一一对应。
\end{theorem}
\begin{proof}
    切矢场逐点满足线性性及 Leibniz 法则，且 $v(f)=v^i\p_i f$ 光滑，故给出导子。

    反之，对导子 $D$ 置 $v_p(f):=D(f)(p)$。由前文全局函数与函数芽的等价定义，$v_p$ 是唯一确定的切矢。取坐标域 $U$ 及光滑函数 $\chi$，使 $\supp\chi\subset U$，且在 $p$ 的某邻域 $V$ 上为 $1$。将 $\chi x^i$ 在域外置零，得全局光滑函数 $F_i$。在 $V$ 上有 $v^i=v(x^i)=D(F_i)$，故各分量光滑。唯一性由每点的唯一性即得。
\end{proof}

\begin{proposition}[局部性]\label{chdm:thm_vf-local}
    若 $f,g\in C^\infty(M)$ 在开集 $U$ 上相等，则任意光滑切矢场 $v$ 满足 $v(f)|_U=v(g)|_U$。
\end{proposition}
\begin{proof}
    对每个 $p\in U$，$f,g$ 在 $p$ 处的函数芽相同，故 $v_p(f)=v_p(g)$。
\end{proof}

\begin{remark}\label{chdm:rmk_local}
    因此全局导子 $D$ 可限制到任意开集 $U$：对 $f\in C^\infty(U)$ 及 $p\in U$，取全局光滑函数 $F$ 在 $p$ 附近与 $f$ 相等，置
    \eq{\label{chdm:eqn_aflg}
        (D|_U)(f)(p):=D(F)(p).
    }
    局部性保证结果与延拓无关。它说明求导只依赖该点附近的函数值，从而可以合法地将全局导子作用于局部坐标函数。
\end{remark}

\begin{definition}
    置 $C^1$-曲线 $\gamma:I\to M$。其在 $p=\gamma(t)$ 处的\textbf{切矢} $\dot\gamma(t)\in T_pM$ 定义为
    \eq{\label{chdm:eqn_DiffCurve}
        \dot\gamma(t)(f):=\dv{(f\circ\gamma)}{t},\quad f\in C^\infty_p(M).
    }
    由链式法则，其坐标展开为 $\dot\gamma(t)=\dv{(x^i\circ\gamma)}{t}\p_i|_{\gamma(t)}$，时常简记 
    \eq{\label{chdm:eqn_DiffCurve-2}
        \dv{t} = \dv{x^i}{t}\p_i.
    }
    映射 $t\mapsto\dot\gamma(t)$ 是沿 $\gamma$ 的\textbf{曲线切矢场}。曲线在 $p$ 处的切矢属于 $T_pM$，而 $T_pM$ 中的切矢未必都切于该曲线。
\end{definition}

\begin{example}
    第 $x^i$ 坐标线的切矢正是坐标基 $\delta^j_i\p_j=\p_i$。
\end{example}


\begin{definition}
    置映射 $f,g:A\to A$。若 $f\circ g=g\circ f$，则称 $f,g$ \textbf{对易}(commute)。若 $A$ 为加法交换群，则可逐点定义其\textbf{对易子}(commutator)
    \[
        [f,g](x):=f(g(x))-g(f(x)).
    \]
    此时 $f,g$ 对易 $\iff[f,g]=0$，其中 $0$ 表示零映射。
\end{definition}

\begin{remark}
    此定义不要求 $f,g$ 线性，但一般映射的对易子未必具有双线性或满足 Jacobi 恒等式。在实线性空间的线性自映射上，对易子具有上述性质，构成通常的 Lie 括号。
\end{remark}

以下 $AB$ 指切矢场作为算子的复合，即 $AB(f)=A(B(f))$。矢量场的对易子称为\textbf{Lie 括号}，亦称\textbf{Poisson 括号}：
\eq{\label{chdm:def_poisson-bracket}
    [A,B](f):=A(B(f))-B(A(f)).
}

\begin{example}
    微积分 Schwarz 定理指出，光滑函数的混合偏导可交换：$ [\p_\mu,\p_\nu ]=0$，称为\textbf{偏导对易性}。
\end{example}

\begin{theorem}
    光滑矢量场对易子 $[A,B]$ 仍是光滑矢量场。
\end{theorem}
\begin{proof}
    $[A,B]$ 显然线性。展开 $A(B(fg))-B(A(fg))$，交叉项抵消，得
    \[
        [A,B](fg)=g[A,B](f)+f[A,B](g).
    \]
    故在每点取值均为切矢。又在坐标域中，二阶偏导项相消，得
    \eq{\label{chdm:eqn_poisson-bracket-expand}
        [A,B]^\lambda=A^\mu\partial_\mu B^\lambda-B^\mu\partial_\mu A^\lambda.
    }
    各分量均光滑，故 $[A,B]$ 为光滑切矢场。如果只使用分量语言，还需额外再证它满足张量变换式：
    \[
        \pdv{x^\sigma}(\pdv{x'^\lambda}{x^\kappa}B^\kappa)=  B^\kappa\pdv{x'^\lambda}{x^\sigma}{x^\kappa}+\pdv{x'^\lambda}{x^\kappa}\partial_\sigma B^\kappa,
    \]
    因为 $\partial_\sigma\partial_\kappa=\partial_\kappa\partial_\sigma$ 始终使 $\pdv{x'^\lambda}{x^\sigma}{x^\kappa}$ 项抵消。
\end{proof}

\begin{theorem}\label{chdm:thm_poisson-Lie-bracket}
    光滑切矢场的 Lie 括号为 $\R$-双线性、反对称，并满足 Jacobi 恒等式：
    \[
        [u,[v,w]]+[v,[w,u]]+[w,[u,v]]=0.
    \]
    因而 $\mathfrak X(M)$ 关于此括号构成实 Lie 代数。
\end{theorem}
\begin{proof}
    双线性与反对称性由定义即得。将 Jacobi 恒等式中的每项展开为三个算子的复合，各项两两抵消。
\end{proof}

Lie 括号一般不具有 $C^\infty(M)$-线性。对光滑函数 $f,g$，由 Leibniz 法则展开其对任意函数的作用，得
\[
    [fu,gv]=fg[u,v]+f\,u(g)v-g\,v(f)u.
\]

\begin{definition}
    $T^*_p M:=(T_p M)^*$ 称为 $M$ 在 $p$ 处的\textbf{余切空间}，其元素称为 $M$ 在 $p$ 处的\textbf{余切矢}。
    全体余切空间的无交并 $T^*M:=\bigsqcup_{p\in M} T^*_p M$ 称为\textbf{余切丛}。
\end{definition}

Cartan 不仅提出了切矢的泛函导子定义，他还注意到，同一关系也可反过来看：切矢 $v$ 作用于函数 $f$ 给出实数 $v(f)$，而固定 $f$，便得到关于 $v$ 的线性泛函。
这相当于给函数微分一个更严格的理解。标准分析学中常将 $\d f$ 定义为 $\Delta f$ 的线性主部，即 $\d f(x,\Delta x):=f'(x)\Delta x$。这里 $\Delta x$ 是从某点 $x$ 出发的位移，作为线性映射的自变量可取任意大小；只有用 $\d f$ 近似实际增量 $\Delta f$ 时，才要求 $\Delta x$ 足够小。物理上常将其理解为微小位移所引起的微小增量。
在流形上，两点不能直接作差，但沿曲线的位移仍可由切矢作一阶描述。因此，将 $\Delta x$ 换为 $p$ 处的切矢 $v$，用方向导数 $v(f)$ 给出微分的作用。它对 $v$ 线性，故属于 $T_p^*M$，这便是微分与余切矢的关系。

\begin{definition}
    置开集 $U\subset M$ 及 $f\in C^\infty(U)$。$f$ 在 $p\in U$ 处的\textbf{微分}为
    \eq{\label{chdm:eqn_dfv}
        (\d f)_p(v):=v(f),\quad v\in T_pM.
    }
    它只依赖 $f$ 在 $p$ 处的函数芽。各点的微分组成余切矢场 $\d f$。
\end{definition}

\begin{theorem}
    坐标函数的微分 $\{(\d x^i)_p\}$ 是 $\{\p_i|_p\}$ 的对偶基，故 $\dim T_p^*M=\dim T_pM=n$。
\end{theorem}
\begin{proof}
    由定义，
    \eq{\label{chdm:eqn_dual-natural-bases}
        (\d x^j)_p(\p_i|_p)=\p_i|_p(x^j)=\delta_i^j.
    }
    故上述两组基对偶，维数结论随即成立。
\end{proof}

$\{(\d x^i)_p\}$ 称为\textbf{对偶坐标基}。以下略去取值点 $p$。任给 $\omega\in T_p^*M$ 及 $v\in T_pM$，有
\[
    \omega=\omega_i\d x^i,\quad \omega_i:=\omega(\p_i),\qquad
    \langle\omega,v\rangle:=\omega(v)=\omega_i v^i.
\]
特别地，$(\d f)(\p_i)=\p_i f$，故
\eq{\label{chdm:eqn_df}
    \d f=\p_i f\,\d x^i.
}
这与通常的全微分一致\footnote{在标准分析中，全微分理解为各线性映射间的关系，这里将线性映射叙述为余切矢量。为区分于标准分析学中的概念，部分书籍会用粗体表示这些泛函，如 $\bm A,\bm\omega,\bm{\partial}/{\bm{\partial}x^i},\mathbf{d}f$。但其实没有必要，因为两种概念其实是一致的。看似不同是因为本科教学为了通俗性，默认以坐标空间为基础，而没有额外提出流形。}。微分本身是余切矢；将其对应为梯度切矢还需指定度规。

由切矢的线性性及 Leibniz 法则，微分满足
\[
    \d(af+bg)=a\,\d f+b\,\d g,\qquad
    \d(fg)=g\,\d f+f\,\d g,\quad a,b\in\R.
\]

\begin{theorem}[余切分量变换律]
    置同一点附近的两套坐标 $\{x\},\{x'\}$，略去取值点。对偶坐标基及同一余切矢的分量满足
    \eq{\label{chdm:eqn_xy-cov-transform}
        \d x'^j=\pdv{x'^j}{x^i}\d x^i,\qquad
        \omega'_j=\pdv{x^i}{x'^j}\omega_i.
    }
\end{theorem}
\begin{proof}
    在式 \eqref{chdm:eqn_df} 中取 $f=x'^j$，即得对偶基变换。将其代入 $\omega=\omega_i\d x^i=\omega'_j\d x'^j$，比较系数并用逆 Jacobi 矩阵，即得分量变换。
\end{proof}

\begin{definition}\label{chdm:def_cotangent-vector-field}
    在每点 $p$ 指定 $\omega_p\in T_p^*M$，称为\textbf{余切矢场}。若局部表达 $\omega=\omega_i\d x^i$ 的分量均光滑，则称为光滑余切矢场，亦称\textbf{一次微分形式}。全体记为 $\Omega^1(M)=\Gamma(T^*M)$，也沿用记号 $\mathfrak X^*(M)$ 或 $A^1(M)$；这里的星号不指实线性空间 $\mathfrak X(M)$ 的全对偶。
\end{definition}

余切丛以 $(x^i,\omega_i)$ 为局部坐标，坐标变换由余切分量变换律给出，其光滑结构与切丛同理。

有了切空间 $T_pM$ 及其对偶空间 $T_p^*M$，只需在 \S\ref{sec:tensor} 中取 $V=T_pM$，便得到流形在 $p$ 处的张量。

\begin{definition}
    $M$ 在 $p$ 处的 \textbf{$(k,l)$-张量}是张量空间
    \eq{\label{chdm:eqn_TM-TsM}
        \mathcal T^k_l(p):=(T_pM)^k_l
        =\underbrace{T_pM\otimes\cdots\otimes T_pM}_{k\text{个}}
        \otimes\underbrace{T_p^*M\otimes\cdots\otimes T_p^*M}_{l\text{个}}
    }
    中的元素，即接受 $k$ 个余切矢及 $l$ 个切矢的实值多重线性函数。由前文知，其维数为 $n^{k+l}$，其中 $n=\dim M$。
\end{definition}

取 $p$ 附近的坐标 $\{x\}$，以 $\p_i|_p$ 及其对偶基 $(\d x^j)_p$ 作张量积，便得到相应的张量基。略去取值点，任意 $\alpha\in\mathcal T^k_l(p)$ 均可唯一展开为
\eq{\label{chdm:eqn_tensor-component}
    \alpha=\alpha^{i_1\cdots i_k}{}_{j_1\cdots j_l}
    \p_{i_1}\otimes\cdots\otimes\p_{i_k}\otimes
    \d x^{j_1}\otimes\cdots\otimes\d x^{j_l}.
}
坐标变换时，将式 \eqref{chdm:eqn_xy-transform}、\eqref{chdm:eqn_xy-cov-transform} 逐因子代入即可，即得熟悉的张量变换式
\[
    {\alpha'}^{k_1 \cdots k_p}_{\phantom{k_1 \cdots k_p}l_1\cdots l_q} (x')
    ={\alpha}^{i_1 \cdots i_p}_{\phantom{i_1 \cdots i_p}j_1\cdots j_q} (x) 
    \pdv{x'^{k_1}}{x^{i_1}} \cdots
    \pdv{x'^{k_p}}{x^{i_p}}
    \pdv{x^{j_1}}{x'^{l_1}} \cdots
    \pdv{x^{j_q}}{x'^{l_q}}.
\]
所有系数均在同一点 $p$ 取值（等式两边分别取 $p$ 的坐标值 $x',x$）。
将 $p$ 遍历定义域就是张量场的变换。

\begin{definition}\label{chdm:def_tensor-field}
    在每点 $p\in M$ 指定 $T_p\in\mathcal T^k_l(p)$，便得到一个\textbf{$(k,l)$-张量场}。若其在各坐标域中的分量均为 $C^r$ 函数，则称为 $C^r$ 张量场；此性质由张量变换律知与坐标选择无关。

    置张量丛 $T^k_lM:=\bigsqcup_{p\in M}\mathcal T^k_l(p)$，光滑张量场即其光滑截面，全体记为 $\mathfrak T^k_l(M)=\Gamma(T^k_lM)$。张量丛的光滑结构由底点坐标及张量分量给出，与切丛的构造相同。
\end{definition}

加法、函数乘法、张量积及缩并均逐点定义，并保持光滑性。因此 $\mathfrak T^k_l(M)$ 是 $C^\infty(M)$-模，且
\[
    \mathfrak T^0_0(M)=C^\infty(M),\quad
    \mathfrak T^1_0(M)=\mathfrak X(M),\quad
    \mathfrak T^0_1(M)=\Omega^1(M).
\]
各型张量场的直和 $\mathfrak T(M):=\bigoplus_{k,l\geqslant0}\mathfrak T^k_l(M)$ 关于张量积构成 $C^\infty(M)$-代数。度规是其中一类特殊的 $(0,2)$-张量场。

切矢场可由其对函数的作用刻画；类似地，张量场可由其对矢量场及余切矢场的作用刻画。关键是对每个位置具有函数线性性，而不只是实线性性。

\begin{theorem}[张量场特征引理]\label{chdm:thm_Tensor-Characterization-Lemma}
    光滑 $(k,l)$-张量场与下述 $C^\infty(M)$-多重线性映射一一对应：
    \[
        \mathscr T:\Omega^1(M)^k\times\mathfrak X(M)^l\to C^\infty(M).
    \]
    对应关系为逐点取值
    \eq{\label{chdm:eqn_Tensor-Characterization-Lemma}
        \begin{aligned}
            &\mathscr T(\alpha_1,\ldots,\alpha_k;v_1,\ldots,v_l)(p)\\
            &\quad=T_p((\alpha_1)_p,\ldots,(\alpha_k)_p;(v_1)_p,\ldots,(v_l)_p).
        \end{aligned}
    }
\end{theorem}
\begin{proof}
    张量场给出的映射显然具有函数线性性，坐标展开保证其值光滑。

    反之，先证 $\mathscr T$ 在 $p$ 处的值只依赖各个场在 $p$ 的取值。固定其余位置，若某个场 $s$ 在 $p$ 附近为零，取支撑含于该邻域且 $\chi(p)=1$ 的光滑函数，则
    \[
        \mathscr T(\ldots,s,\ldots)(p)
        =\mathscr T(\ldots,\chi s,\ldots)(p)=0.
    \]
    故该值只依赖场在 $p$ 附近的限制。若仅有 $s_p=0$，在局部标架中写 $s=s^i e_i$，其中 $s^i(p)=0$。将系数及基场乘以在 $p$ 附近为 $1$ 的隆起函数，再在坐标域外置零，便可用全局场计算。由局部性及函数线性性，
    \[
        \mathscr T(\ldots,s,\ldots)(p)
        =s^i(p)\mathscr T(\ldots,e_i,\ldots)(p)=0.
    \]
    每个切矢或余切矢都可如此延拓成光滑场，故上式唯一确定逐点张量 $T_p$。在一小坐标邻域内，将坐标基及对偶基延拓成全局场，代入 $\mathscr T$ 所得的光滑函数正是 $T$ 的分量，故 $T$ 光滑。
\end{proof}

\begin{corollary}\label{chdm:thm_Tensor-Characterization-Lemma-1}
    $C^\infty(M)$-多重线性映射 $\Omega^1(M)^k\times\mathfrak X(M)^l\to\mathfrak X(M)$ 唯一对应一个光滑 $(k+1,l)$-张量场。
\end{corollary}
\begin{proof}
    再用一个余切矢场作用于输出，便得到实函数值的多重线性映射，应用上述引理即可。
\end{proof}

前文在线性空间上定义的度规，逐点赋予切空间并要求光滑，便得到流形上的度规场。

\begin{definition}
    $M$ 上对称、处处非退化且号差固定的光滑 $(0,2)$-张量场 $g$ 称为\textbf{度规场}，简称\textbf{度规}。赋有度规的 $(M,g)$ 称为\textbf{伪 Riemann 流形}；其中 $g$ 正定时称为\textbf{Riemann 流形}。
\end{definition}

度规给出各切空间上的内积 $g_p(v,w)$。在坐标系中，
\eq{g=g_{ij}\,\d x^i\otimes\d x^j,\qquad g_{ij}=g(\p_i,\p_j),\qquad \det(g_{ij})\ne0.}
音乐同构及升降指标均逐点沿用线性空间的定义；矩阵求逆保持光滑，故 $g^{-1}$ 也是光滑张量场。Sylvester 惯性定理保证号差无关坐标选择；光滑性与非退化性又使惯性指数局部恒定，故 $M$ 连通时无需另加号差固定的要求。

\begin{definition}
    规范矩阵中只有一个负号或只有一个正号的不定度规称为 \textbf{Lorentz 度规}。本书取一个负方向，即规范矩阵为 $\diag(-1,1,\cdots,1)$；四维时号差为 $+2$，记
    \eq{(\eta_{\mu\nu})=\diag(-1,1,1,1).}
\end{definition}

\begin{remark}
    在任意指定点 $p$，可对坐标作线性变换，使 $(g_{ij}(p))=\diag(\pm1,\cdots,\pm1)$。这只保证在该点成立，一般不能使整个坐标域上的度规分量都成为同一常矩阵。局部正交归一标架也未必是坐标标架。
\end{remark}

\begin{theorem}
    每个光滑流形都存在 Riemann 度规。
\end{theorem}
\begin{proof}
    取坐标开覆盖及从属的局部有限光滑单位分解 $\{\rho_\alpha\}$。各坐标域上置 $g_\alpha:=\sum_i\d x_\alpha^i\otimes\d x_\alpha^i$，将 $\rho_\alpha g_\alpha$ 在域外置零。则 $g:=\sum_\alpha\rho_\alpha g_\alpha$ 光滑、对称；对任意非零切矢 $v$，有 $g(v,v)>0$，故为 Riemann 度规。
\end{proof}

\begin{definition}
    分段 $C^1$ 曲线 $C:[a,b]\to M$ 的\textbf{线长}为
    \eq{L(C):=\int_a^b\sqrt{\abs{g_{C(t)}(\dot C(t),\dot C(t))}}\,\d t,}
    其中 $\dot C(t)=\dv*{}{t}$ 为曲线切矢。在坐标中，被积函数为 $\sqrt{\abs{g_{ij}(C(t))\dot x^i(t)\dot x^j(t)}}$。
\end{definition}

\begin{theorem}
    线长在分段 $C^1$ 微分同胚的重新参数化下不变，包括参数方向的反转。
\end{theorem}
\begin{proof}
    若 $t=\phi(s)$、$\widetilde C=C\circ\phi$，则 $\dot{\widetilde C}=\phi'\dot C\circ\phi$，故
    \eq{\sqrt{\abs{g(\dot{\widetilde C},\dot{\widetilde C})}}
        =|\phi'|\sqrt{\abs{g(\dot C,\dot C)}}\circ\phi.}
    对积分作变量替换即得。
\end{proof}

\begin{remark}
    曲线允许停留或自交：停留段贡献为零，重复经过同一路段则重复计长。因此线长并非曲线像集的测度。不定度规下，非零切矢也可能满足 $g(v,v)=0$，故非恒定曲线的线长也可能为零。
\end{remark}

若 $C$ 为 $C^1$ 曲线且 $g(\dot C,\dot C)$ 处处非零，则累积线长
\eq{\sigma(t):=\int_a^t\sqrt{\abs{g_{C(u)}(\dot C(u),\dot C(u))}}\,\d u}
严格递增，可作\textbf{线长参数}。由链式法则，
\eq{g\left(\dv{}{\sigma},\dv{}{\sigma}\right)
    =\frac{g(\dot C,\dot C)}{|g(\dot C,\dot C)|}=\pm1.}

\begin{example}
    对采用本书号差约定的 Lorentz 度规，类时曲线满足 $g(\dot C,\dot C)<0$，其线长就是\textbf{固有时}（取 $c=1$）：
    \eq{\tau(t)-\tau(a)=\int_a^t\sqrt{-g_{C(u)}(\dot C(u),\dot C(u))}\,\d u.}
    以固有时为参数的切矢 $U:=\dv*{}{\tau}$ 满足 $g(U,U)=-1$。类光曲线的切矢非零但模方为零，故不能用线长参数化。
\end{example}

\begin{definition}
    $\R^n$ 配以自然坐标中的度规 $\delta:=\delta_{ij}\,\d x^i\otimes\d x^j$，称为\textbf{欧氏空间} $\mathbb E^n:=(\R^n,\delta)$。使度规分量恒为 $\delta_{ij}$ 的坐标系称为 \textbf{Descartes 坐标系}。

    $\R^4$ 配以自然坐标中的度规 $\eta:=\eta_{\mu\nu}\,\d x^\mu\otimes\d x^\nu$，称为\textbf{闵氏时空} $\R^{1,3}:=(\R^4,\eta)$。使度规分量恒为 $\eta_{\mu\nu}$ 的坐标系称为 \textbf{Lorentz 坐标系}，亦称\textbf{惯性坐标系}。
\end{definition}

\begin{theorem}
    在连通的重叠域内，两 Descartes 坐标系之间的变换必为
    \eq{x'=Qx+a,\qquad Q^{\mathsf T}Q=I;}
    两 Lorentz 坐标系之间的变换必为
    \eq{x'=\Lambda x+a,\qquad \Lambda^{\mathsf T}\eta\Lambda=\eta.}
    其中矩阵及平移量均为常量。反之，这些变换均保持相应的度规分量。
\end{theorem}
\begin{proof}
    将两种常度规矩阵统记为 $G$。设坐标变换 $x'=F(x)$，置 $F_i:=\p_iF$、$F_{ij}:=\p_i\p_jF$，并以 $\langle\ ,\ \rangle_G$ 表示 $G$ 给出的双线性函数。度规变换律给出 $\langle F_i,F_j\rangle_G=G_{ij}$。
    记 $A_{ijk}:=\langle F_{ij},F_k\rangle_G$。由偏导对易及上式求导，分别得 $A_{ijk}=A_{jik}$、$A_{ijk}=-A_{ikj}$，于是
    \eq{A_{ijk}=A_{jik}=-A_{jki}=-A_{kji}=A_{kij}=A_{ikj}=-A_{ijk}.}
    故 $A_{ijk}=0$。因 $\{F_k\}$ 为一组基且 $G$ 非退化，知 $F_{ij}=0$。于是 Jacobi 矩阵在连通域内为常量，$F$ 为仿射变换；代回度规变换律即得矩阵条件及其逆命题。
\end{proof}

因此 Descartes 坐标变换由平移及正交变换组成，后者包括旋转与反射；惯性坐标变换则由平移及 Lorentz 变换组成，合称 Poincar\'e 变换。这里 $\Lambda$ 保持的是 $\eta$，一般不满足 $\Lambda^{\mathsf T}\Lambda=I$。

\section{诱导映射}\label{chdm:sec_induced-map}
\label{chdm:sec_induced-map-on-fields}\label{chdm:sec_related}

置光滑映射 $\phi:M\to N$。给定标量场 $f:N\to\R$，$p\in M$ 变到 $\phi(p)\in N$ 后，标量场的值为 $f(\phi(p))$。我们可将这个值直接搬回 $p$ 处，得到 $M$ 上的新标量场 $\phi^*f$，满足 $\phi^*f(p)=f(\phi(p))$。这就是\textbf{拉回}：点沿 $\phi$ 向前走，函数值沿 $\phi$ 搬回来。

类似地，若 $\phi$ 是微分同胚，则也可将 $p=\phi^{-1}(q)$ 处的函数值搬至 $q$ 处，即对 $h\in C^\infty(M)$ 定义 $\phi_*h:=h\circ\phi^{-1}$。这里需要光滑逆映射，才能保证得到的函数仍光滑。一般映射没有这样的逆：若不同点被映到同一点，其函数值可能不同；不在像中的点也无值可搬。

切矢通过作用于函数来定义，因此函数的拉回也将诱导切矢的推前。再对切映射取线性对偶，便得到余切矢的拉回。以下置 $p\in M$、$q:=\phi(p)$。

这与附录 \ref{appx:cat} 的 $\hom$-函子直接相连。因为 $C^\infty(M)=\hom_{\cate{Man}}(M,\R)$，故标量函数的拉回正是定义 \ref{def:hom-functor} 中固定目标 $\R$、对来源作预复合的构造。其逆变性表现为 $\phi:M\to N$ 诱导反向的 $\phi^*:C^\infty(N)\to C^\infty(M)$。

\begin{definition}\label{chdm:def_pullback-scalar}\label{chdm:def_pullback-scalar-onfield}
    标量函数的\textbf{拉回}为
    \[
        \phi^*:C^\infty(N)\to C^\infty(M),\qquad \phi^*f:=f\circ\phi.
    \]
    同样的公式给出函数芽的拉回 $\phi^*:C^\infty_q(N)\to C^\infty_p(M)$。它是良定义的：在 $q$ 附近相等的函数，复合 $\phi$ 后在 $p$ 附近仍相等。
\end{definition}

由逐点运算知，拉回保持线性性及乘法，即
\[
    \phi^*(af+bg)=a\phi^*f+b\phi^*g,\qquad
    \phi^*(fg)=(\phi^*f)(\phi^*g),\quad a,b\in\R.
\]

\begin{definition}\label{chdm:def_pushforward-vector}
    $\phi$ 在 $p$ 处的\textbf{推前映射}，亦称\textbf{切映射}或\textbf{微分}，为
    \[
        \phi_{*p}=\d\phi_p:T_pM\to T_qN,\qquad
        (\phi_{*p}v)(f):=v(\phi^*f),\quad f\in C^\infty_q(N).
    \]
    取值点明确时简记 $\phi_*v$。
\end{definition}

\begin{proposition}
    推前 $\phi_*v$ 是 $q$ 处的切矢，且 $\phi_{*p}$ 为线性映射。
\end{proposition}
\begin{proof}
    $\phi_*v$ 的线性性由 $v$ 及 $\phi^*$ 的线性性给出。又
    \[
        (\phi_*v)(fg)=v\bigl((\phi^*f)(\phi^*g)\bigr)
        =g(q)(\phi_*v)(f)+f(q)(\phi_*v)(g),
    \]
    故满足 Leibniz 法则。对任意切矢 $u,v$ 及 $a,b\in\R$，定义又直接给出 $\phi_*(au+bv)=a\phi_*u+b\phi_*v$。
\end{proof}

这也落实了例 \ref{eg:double-dual-functor} 中“先拉回输入，再推前泛函”的构造：$\phi_*v=v\circ\phi^*$。准确地说，函数芽的拉回是线性映射，其对偶将 $C^\infty_p(M)$ 上的线性泛函送到 $C^\infty_q(N)$ 上；上述命题保证它将点导子仍送到点导子，限制于切空间便得到 $\phi_{*p}$。这里的切空间是满足 Leibniz 法则的导子子空间，并非全体泛函。

\begin{definition}\label{chdm:def_pullback-1form}
    $\phi$ 在 $p$ 处的\textbf{余切映射}是切映射的对偶，亦称余切矢的\textbf{拉回}：
    \[
        \phi_p^*:T_q^*N\to T_p^*M,\qquad
        (\phi_p^*\omega)(v):=\omega(\phi_{*p}v).
    \]
    取值点明确时简记 $\phi^*\omega$。
\end{definition}

这正是前文对偶函子在 $\phi_{*p}$ 上的作用，即 $\phi_p^*=(\phi_{*p})^*$，故余切映射也线性。函数拉回与余切矢拉回都记作 $\phi^*$，由作用对象区分。切映射与余切映射的起点、终点分别是 $p,q$ 处的切空间及其对偶空间。

一般不能反向拉回切矢：给定 $w\in T_qN$，方程 $\phi_*v=w$ 可能无解，也可能有多个解。若 $\phi$ 为微分同胚，则可用 $(\phi^{-1})_*$ 拉回切矢，用 $(\phi^{-1})^*$ 推前余切矢。

\begin{remark}
    对实函数 $f:M\to\R$，以 $s$ 为 $\R$ 上的自然坐标，有
    \[
        f_{*p}v=v(f)\left.\pdv{s}\right|_{f(p)}.
    \]
    将 $T_{f(p)}\R$ 通过自然坐标基识别为 $\R$，便得到前文的微分 $(\d f)_p(v)=v(f)$。因此标量函数的微分是切映射的特例。
\end{remark}

取 $M,N$ 的坐标系 $(U,\psi),(V,\eta)$，坐标分别为 $\{x^i\},\{y^\alpha\}$，使 $p\in U$、$\phi[U]\subset V$。如 \S\ref{sec:manifold}，将 $\phi$ 写成坐标表达 $y^\alpha=\phi^\alpha(x)$，其中 $\phi^\alpha:=y^\alpha\circ\phi\circ\psi^{-1}$。置
\[
    J^\alpha{}_i(p):=\pdv{\phi^\alpha}{x^i}\bigg|_{\psi(p)}
    =\p_i|_p(y^\alpha\circ\phi).
\]
$J$ 是 $n\times m$ 的 Jacobi 矩阵，其中 $m=\dim M,n=\dim N$，两者不必相等。

\begin{proposition}\label{chdm:thm_coortrans}
    切映射及余切映射在坐标基上的作用为
    \eq{\label{chdm:eqn_push-bases}
        \phi_{*p}(\p_i|_p)=J^\alpha{}_i(p)\left.\pdv{y^\alpha}\right|_q,
    }
    \eq{\label{chdm:eqn_pull-bases}
        \phi_p^*((\d y^\alpha)_q)=J^\alpha{}_i(p)(\d x^i)_p.
    }
\end{proposition}
\begin{proof}
    切矢的坐标分量由其作用于坐标函数给出，故
    \[
        \bigl(\phi_*(\p_i|_p)\bigr)(y^\alpha)
        =\p_i|_p(y^\alpha\circ\phi)=J^\alpha{}_i(p).
    \]
    再由对偶关系，$(\phi^*(\d y^\alpha)_q)(\p_i|_p)=J^\alpha{}_i(p)$，即得第二式。
\end{proof}

因此，略去取值点，任意切矢及余切矢的分量满足
\[
    (\phi_*v)^\alpha=J^\alpha{}_i v^i,\qquad
    (\phi^*\omega)_i=J^\alpha{}_i\omega_\alpha.
\]
若均以列表示分量，则切映射的矩阵为 $J$，余切映射的矩阵为 $J^{\mathsf T}$。取 $N=M$、$\phi=\id_M$，而在两端选用不同坐标，便恢复前文的坐标基变换式。

\begin{proposition}
    微分与拉回可交换：对 $N$ 上的光滑函数 $f$，有
    \eq{\label{chdm:eqn_dfphi}
        \phi^*(\d f)=\d(\phi^*f).
    }
\end{proposition}
\begin{proof}
    对任意 $p\in M$ 及 $v\in T_pM$，
    \[
        \bigl(\phi_p^*(\d f)_q\bigr)(v)
        =(\d f)_q(\phi_*v)=v(f\circ\phi)
        =\bigl(\d(\phi^*f)\bigr)_p(v).
    \]
    因而两者相等。
\end{proof}

特别地，坐标微分的拉回为
\[
    \phi_p^*((\d y^\alpha)_q)
    =\bigl(\d(y^\alpha\circ\phi)\bigr)_p
    =J^\alpha{}_i(p)(\d x^i)_p.
\]
所以，$J$ 的第 $\alpha$ 行给出目标坐标微分拉回后的分量，第 $i$ 列给出源坐标基矢推前后的分量。两者由同一组偏导数联系，这就是余切映射、微分与 Jacobi 矩阵的关系。

\begin{theorem}[链式法则]\label{chdm:thm_map-chain-rule}
    置光滑映射 $\phi:M\to N$、$\psi:N\to Q$，并记 $q=\phi(p)$。则

    (1) 函数拉回满足 $(\psi\circ\phi)^*=\phi^*\circ\psi^*$；

    (2) 切映射满足 $(\psi\circ\phi)_{*p}=\psi_{*q}\circ\phi_{*p}$；

    (3) 余切映射满足 $(\psi\circ\phi)^*_p=\phi_p^*\circ\psi_q^*$。
\end{theorem}
\begin{proof}
    第 (1) 式即复合的结合律：$f\circ(\psi\circ\phi)=(f\circ\psi)\circ\phi$。对任意 $v\in T_pM$ 及 $\psi(q)$ 处的函数芽 $f$，有
    \[
        ((\psi\circ\phi)_*v)(f)=v(f\circ\psi\circ\phi)
        =(\psi_*(\phi_*v))(f),
    \]
    即得第 (2) 式。第 (3) 式由第 (2) 式取线性对偶即得。
\end{proof}

恒等映射也诱导相应的恒等映射，故上述构造满足 \S\ref{sec:functors} 的函子公理。具体地，以带基点的流形 $(M,p)$ 为对象、保持基点的光滑映射为态射，构成 $\cate{Man}_\bullet$。则 $(M,p)\mapsto T_pM$、$\phi\mapsto\phi_{*p}$ 给出协变函子 $\cate{Man}_\bullet\to\cate{Vect}(\R)$；取线性对偶，$(M,p)\mapsto T_p^*M$、$\phi\mapsto\phi_p^*$ 则给出逆变函子。函数拉回的逆变性同样由第 (1) 式给出。

\begin{example}\label{chdm:exm_phiD=Dphi}
    对光滑曲线 $\gamma:I\to M$，其切矢就是切映射作用于参数方向的结果：
    \[
        \dot\gamma(t)=\gamma_{*t}\left(\left.\pdv{t}\right|_t\right),
    \]
    因两边作用于 $f$ 都给出 $\dv{(f\circ\gamma)}{t}$。反之，任给 $v=v^i\p_i|_p$，在坐标域内取
    \[
        \gamma(t):=\psi^{-1}\bigl(\psi(p)+t(v^1,\ldots,v^n)\bigr)
    \]
    （$|t|$ 足够小），便有 $\gamma(0)=p$、$\dot\gamma(0)=v$。因此每个切矢都可由曲线的切映射得到。

    再由链式法则，曲线切矢的推前等于像曲线的切矢：
    \eq{\label{chdm:eqn_phiD=Dphi}
        \phi_{*\gamma(t)}\dot\gamma(t)=\dv{(\phi\circ\gamma)}{t}.
    }
\end{example}

\begin{example}
    置 $f:\R\to\R^2$、$f(t)=(t,t^2)$，及 $g:\R^2\to\R$、$g(x,y)=x\cos y$。在自然坐标基下，切映射的矩阵为
    \[
        [f_{*t}]=\begin{pmatrix}1\\2t\end{pmatrix},\qquad
        [g_{*(x,y)}]=\begin{pmatrix}\cos y&-x\sin y\end{pmatrix}.
    \]
    因 $(f\circ g)(x,y)=(x\cos y,x^2\cos^2y)$，由链式法则，
    \[
        [(f\circ g)_{*(x,y)}]
        =[f_{*g(x,y)}][g_{*(x,y)}]
        =\begin{pmatrix}
            \cos y&-x\sin y\\
            2x\cos^2y&-2x^2\cos y\sin y
        \end{pmatrix}.
    \]
    将 $f$ 看作曲线，其切矢为 $\dot f(t)=\p_x|_{f(t)}+2t\p_y|_{f(t)}$。用 $s$ 表示目标实直线的坐标，则
    \[
        g_{*f(t)}\dot f(t)
        =\bigl(\cos(t^2)-2t^2\sin(t^2)\bigr)\left.\pdv{s}\right|_{t\cos(t^2)}.
    \]
    另一方面，像曲线为 $(g\circ f)(t)=t\cos(t^2)$，直接求导即得同一切矢，验证了式 \eqref{chdm:eqn_phiD=Dphi}。
\end{example}

点上的诱导映射推广到场时，还须照顾取值点。切映射合在一起给出光滑映射 $T\phi:TM\to TN$，其局部表达为 $(x,v)\mapsto(\phi(x),J(x)v)$；但推前一个场后，得到的未必是 $N$ 上的场。

\begin{definition}\label{chdm:def_pullback-1form-onfield}
    对光滑映射 $\phi:M\to N$ 及 $N$ 上的余切矢场 $\omega$，逐点定义其\textbf{拉回}
    \[
        (\phi^*\omega)_p(v):=\omega_{\phi(p)}(\phi_{*p}v).
    \]
    更一般地，对光滑 $(0,l)$-张量场 $T$，置
    \[
        (\phi^*T)_p(v_1,\ldots,v_l)
        :=T_{\phi(p)}(\phi_{*p}v_1,\ldots,\phi_{*p}v_l).
    \]
\end{definition}

这些拉回由坐标公式知均光滑，既不要求 $\phi$ 单射，也不要求满射。标量函数的拉回也是如此：即使 $\phi:\R\to\R$ 恒为 $0$，仍有 $\phi^*f\equiv f(0)$。

对于 $M$ 上的切矢场 $v$，$p\mapsto\phi_{*p}v_p$ 首先只是\textbf{沿 $\phi$ 的矢量场}：每个源点 $p$ 指定 $T_{\phi(p)}N$ 中一个切矢。若 $\phi(p_1)=\phi(p_2)$，两处推前可能不同；而像集以外没有指定取值，故一般不能得到整个 $N$ 上的场。

\begin{definition}\label{chdm:def_pushforward-vector-onfield}
    若 $\phi:M\to N$ 为微分同胚，则切矢场 $v$ 的\textbf{推前}由 $(\phi_*v)_{\phi(p)}:=\phi_{*p}v_p$ 定义。等价地，
    \eq{\label{chdm:eqn_pushforward-vector-onfield}
        (\phi_*v)(f)=\bigl(v(f\circ\phi)\bigr)\circ\phi^{-1},\qquad f\in C^\infty(N).
    }
    切矢场的拉回为 $\phi^*Y:=(\phi^{-1})_*Y$，余切矢场的推前为 $\phi_*\omega:=(\phi^{-1})^*\omega$。
\end{definition}

\begin{definition}\label{chdm:def_push-pull-tensor}
    对微分同胚 $\phi:M\to N$ 及 $M$ 上的 $(k,l)$-张量场 $T$，其\textbf{推前}为
    \[
    \begin{aligned}
        &(\phi_*T)_q(\alpha_1,\ldots,\alpha_k;w_1,\ldots,w_l)\\
        &\quad:=T_p\bigl(\phi_p^*\alpha_1,\ldots,\phi_p^*\alpha_k;
        (\phi^{-1})_{*q}w_1,\ldots,(\phi^{-1})_{*q}w_l\bigr),
    \end{aligned}
    \]
    其中 $q=\phi(p)$、$\alpha_i\in T_q^*N$、$w_j\in T_qN$。对 $N$ 上的张量场 $S$，其\textbf{拉回}定义为 $\phi^*S:=(\phi^{-1})_*S$。这些定义与函数、切矢及余切矢的变换一致。
\end{definition}

\begin{proposition}\label{chdm:thm_fxxf}\label{chdm:thm_pptt}
    微分同胚诱导的推前、拉回保持加法、实数乘法、张量积及缩并，并满足
    \[
    \begin{aligned}
        \phi_*(fT)&=(f\circ\phi^{-1})\phi_*T,&
        \phi^*(gS)&=(g\circ\phi)\phi^*S,\\
        \phi^*(S\otimes R)&=\phi^*S\otimes\phi^*R,&
        (\phi_*v)(f\circ\phi^{-1})&=v(f)\circ\phi^{-1},
    \end{aligned}
    \]
    其中 $f,T,v$ 定义在 $M$ 上，$g,S,R$ 定义在 $N$ 上。
\end{proposition}
\begin{proof}
    前三式及加法、实数乘法的保持由逐点定义即得；最后一式由切矢场推前公式取测试函数 $f\circ\phi^{-1}$ 得到。缩并的保持归结为配对不变：$(\phi_*\omega)(\phi_*v)=\omega(v)\circ\phi^{-1}$。
\end{proof}

微分同胚保证每个矢量场都可推前。对一般光滑映射，也可考察两边已有的场是否由切映射相联系。
\begin{definition}\label{chdm:def_related}
    置光滑映射 $\phi:M\to N$，$X\in\mathfrak X(M),Y\in\mathfrak X(N)$。若对每个 $p\in M$ 有 $\phi_{*p}X_p=Y_{\phi(p)}$，则称 $X,Y$ 为\textbf{$\phi$-相关矢量场}。等价地，对任意 $f\in C^\infty(N)$，
    \[
        X(\phi^*f)=\phi^*(Y(f)).
    \]
\end{definition}

\begin{theorem}\label{chdm:thm_push-Poisson-related}
    若 $X_i,Y_i$ 为 $\phi$-相关矢量场（$i=1,2$），则 $[X_1,X_2]$ 与 $[Y_1,Y_2]$ 也 $\phi$-相关。
\end{theorem}
\begin{proof}
    对任意 $f\in C^\infty(N)$，两次应用相关性，得
    \[
    \begin{aligned}
        [X_1,X_2](\phi^*f)
        &=X_1\bigl(\phi^*(Y_2f)\bigr)-X_2\bigl(\phi^*(Y_1f)\bigr)\\
        &=\phi^*\bigl(Y_1Y_2f-Y_2Y_1f\bigr)
        =\phi^*([Y_1,Y_2]f).
    \end{aligned}
    \]
    即得所求。
\end{proof}

\begin{corollary}\label{chdm:thm_push-Poisson}
    微分同胚保持 Lie 括号：
    \eq{\label{chdm:eqn_push-Poisson}
        \phi_*[X,Y]=[\phi_*X,\phi_*Y].
    }
\end{corollary}
\begin{proof}
    $X,\phi_*X$ 及 $Y,\phi_*Y$ 分别 $\phi$-相关，应用上一定理即可。
\end{proof}

\section{嵌入映射}

切映射描述光滑映射在一点附近的一阶行为。其秩决定哪些方向被保留、哪些方向被压缩；若还要将整个流形视作另一个流形的子空间，则须进一步考察单射性及像集的子拓扑。

置光滑映射 $\phi:M\to N$，$m=\dim M,n=\dim N$。定义 $\phi$ 在 $p$ 处的\textbf{秩}为
\[
    \rank_p\phi:=\rank\phi_{*p}=\rank J(p)\leqslant\min(m,n).
\]
换坐标只会在 $J$ 两侧乘可逆矩阵，故秩与坐标无关。秩未必在每点附近恒定，例如 $t\mapsto t^3$ 在 $0$ 处秩为 $0$，在其余点秩为 $1$。但非零子式在充分小的邻域内仍非零，故最大秩条件可保持到邻域上。

\begin{theorem}[逆函数定理]\label{chdm:thm_inv-in-M}
    若 $m=n$ 且 $\phi_{*p}$ 为同构，则存在 $p$ 的开邻域 $U$ 及 $\phi(p)$ 的开邻域 $V$，使 $\phi|_U:U\to V$ 为微分同胚。
\end{theorem}
\begin{proof}
    取两端的坐标系，将 $\phi$ 表为欧氏开集间的光滑映射。其在 $p$ 的坐标处 Jacobi 矩阵可逆，由通常的逆函数定理即得。
\end{proof}

\begin{definition}
    对光滑映射 $\phi:M\to N$，规定：
    \begin{itemize}
        \item 若每点的 $\phi_{*p}$ 均为单射，即 $\rank_p\phi=m$，则称 $\phi$ 为\textbf{浸入}(immersion)，此时 $m\leqslant n$；
        \item 若 $\phi$ 为浸入，且 $\phi:M\to\phi[M]$ 为同胚，其中 $\phi[M]$ 取 $N$ 的子拓扑，则称 $\phi$ 为\textbf{嵌入}(embedding)，记 $M\hookrightarrow N$；
        \item 若每点的 $\phi_{*p}$ 均为满射，即 $\rank_p\phi=n$，则称 $\phi$ 为\textbf{淹没}(submersion)，此时 $m\geqslant n$。
    \end{itemize}
    若只在某点满足相应的秩条件，则称在该点浸入或淹没。
\end{definition}

浸入和淹没都是局部条件；当 $m=n$ 时，两者都等价于局部微分同胚。嵌入还要求整体单射，并使源空间拓扑与像集的子拓扑一致，这与 \S\ref{sec:manifold} 的定义相同。

\begin{theorem}[浸入的局部标准形]\label{chdm:thm_immerse-1}
    若 $\phi$ 在 $p$ 处浸入，则可取以 $p,\phi(p)$ 为原点的坐标系 $(U,x),(V,y)$，使 $\phi[U]\subset V$，且
    \[
        y\circ\phi\circ x^{-1}(u)=(u^1,\ldots,u^m,0,\ldots,0).
    \]
\end{theorem}
\begin{proof}
    先任选原点位于 $p,\phi(p)$ 的坐标，将 $\phi$ 表为 $F$。重排目标坐标，可使原点处 Jacobi 矩阵的前 $m$ 行组成可逆方阵。令
    \eq{\label{chdm:eqn_tmp88}
        H(u,z):=F(u)+(0,z),\qquad z\in\R^{n-m}.
    }
    $H$ 在 $(0,0)$ 处的 Jacobi 矩阵为分块三角矩阵，其对角块为上述可逆矩阵和单位矩阵，故 $H$ 在原点附近为微分同胚。用 $H^{-1}$ 改换目标坐标，由 $H(u,0)=F(u)$ 得
    \eq{\label{chdm:eqn_tmp98}
        H^{-1}\circ F(u)=(u,0),
    }
    即得所求。
\end{proof}

此式说明，浸入在充分小的源邻域上就是坐标面的包含映射，因而局部上是嵌入。但源邻域之外的点仍可能映入附近，甚至与其重合，故整体上未必单射，也未必为嵌入。

\begin{theorem}[淹没的局部标准形]
    若 $\phi$ 在 $p$ 处淹没，则可取两端的局部坐标，使其表为投影
    \[
        (u^1,\ldots,u^m)\longmapsto(u^1,\ldots,u^n).
    \]
\end{theorem}
\begin{proof}
    将 $\phi$ 表为 $F$，重排源坐标，使原点处 Jacobi 矩阵的前 $n$ 列组成可逆方阵。于是 $H(u):=(F(u),u^{n+1},\ldots,u^m)$ 在原点附近为微分同胚，以 $H$ 为新源坐标即得。
\end{proof}

\begin{definition}\label{chdm:def_inclusion-map}
    对子集 $S\subset N$，映射 $\imath:S\to N$、$\imath(p)=p$ 称为\textbf{包含映射}(inclusion map)。若 $S$ 赋有流形结构，使 $\imath$ 为浸入，则称 $S$ 为 $N$ 的\textbf{浸入子流形}；若 $\imath$ 为嵌入，则称\textbf{嵌入子流形}。
\end{definition}

单射浸入 $\phi:M\to N$ 可将 $M$ 的拓扑及光滑结构搬到 $S=\phi[M]$ 上，使 $\phi:M\to S$ 为微分同胚；此时 $S$ 是浸入子流形。这种拓扑可能严于 $N$ 的子拓扑，两者一致时才是嵌入子流形。因而包含映射虽总是单射，却并非对任意所赋的流形结构都为嵌入。

\begin{example}\label{chdm:exm_opensub}
    \textbf{开子流形}。对开集 $U\subset N$，限制 $N$ 的坐标系便给出 $U$ 的光滑结构。包含映射 $U\hookrightarrow N$ 为嵌入，且 $\dim U=\dim N$。
\end{example}

\begin{example}
    将直线嵌入平面为 $x$ 轴：$\imath:\R\to\R^2$、$\imath(t)=(t,0)$。场 $X=\p_t$ 推前后只给出轴上的切矢 $\p_x$，并未指定轴外的取值。但平面上的 $Y=\p_x$ 或 $Y=\p_x+y\p_y$ 都与 $X$ 为 $\imath$-相关。这说明相关性只约束像集上的取值，未必唯一确定目标流形上的场。
\end{example}

\begin{example}\label{chdm:exm_closesub}
    \textbf{闭子流形}是作为子集闭的嵌入子流形。例如 $S^2\subset\R^3$：在任意点附近，可将一个非零坐标由 $x^2+y^2+z^2=1$ 光滑地解出，故局部为光滑函数的图像；配以子拓扑，其包含映射为嵌入，且 $S^2$ 在 $\R^3$ 中闭。
\end{example}

下面两个相近的平面曲线说明，浸入的局部性质不足以保证嵌入。
\begin{example}\label{chdm:exm_iec1}
    浸入可以有自交。置
    \eq{\label{chdm:eqn_tmp-f-immerse}
        F:\R\to\R^2,\qquad F(t)=(2\sin t,-\sin2t).
    }
    $F'(t)=(2\cos t,-2\cos2t)$ 处处非零，故 $F$ 为浸入。但 $F(0)=F(\pi)=(0,0)$，故不单射，从而不是嵌入。
\end{example}

\begin{example}\label{chdm:exm_iec2}
    单射浸入仍未必是嵌入。置
    \eq{\label{chdm:eqn_tmp-g-embed}
        G(t)=(2\cos\theta(t),\sin2\theta(t)),\qquad
        \theta(t):=2\arctan t+\frac\pi2.
    }
    $\theta$ 严格递增，值域为 $(-\pi/2,3\pi/2)$，且 $\theta'(t)>0$。因 $(-2\sin\theta,2\cos2\theta)$ 处处非零，$G$ 为浸入。第一分量非零时，可由两分量恢复 $\cos\theta$ 及 $\sin\theta$；第一分量为零时，只有 $\theta=\pi/2$。故 $G$ 也单射。

    然而 $G(0)=(0,0)$，且 $G(t)\to(0,0)$（$t\to\pm\infty$）。取 $t_k\to+\infty$，则 $G(t_k)\to G(0)$，而 $t_k$ 不趋于 $0$，故像集取子拓扑时 $G^{-1}$ 不连续，$G$ 不是嵌入。几何上，原点的任意小圆盘都会同时遇到 $t=0$ 附近的曲线和参数远处的尾部；由源空间搬来的拓扑则可将两者分开。
\end{example}

\begin{example}\label{chdm:exm_iec3}
    尾部还可向一整段曲线积聚。置 $H:\R\to\R^2$，在两端定义
    \[
        H(t)=\begin{cases}
            (0,t+2),&t\leqslant-1,\\
            (3/t^2,\sin\pi t),&t\geqslant1.
        \end{cases}
    \]
    在 $-1<t<1$ 上，沿区域 $\{y>1\}\cup\{x>3\}$ 选取无自交且切矢处处非零的光滑弧连接两端，并在端点处与上述两式光滑衔接。两端的切矢也处处非零，且三个部分除接点外互不相交，故 $H$ 为单射浸入。

    其尾部在右侧振荡，向像中 $y$ 轴的 $[-1,1]$ 段积聚。特别地，$H(2k)=(3/(4k^2),0)\to(0,0)=H(-2)$，而 $2k$ 不趋于 $-2$。因此 $H^{-1}$ 对像集的子拓扑不连续，$H$ 不是嵌入。这与上一例具有相同的拓扑障碍。
\end{example}

下面给出保证嵌入的条件，并说明嵌入子流形在环境空间中的局部形状。
\begin{theorem}\label{chdm:thm_embed-1}
    若 $M$ 紧，则任意单射浸入 $\phi:M\to N$ 都是嵌入，且 $\phi[M]$ 为闭子流形。
\end{theorem}
\begin{proof}
    $N$ 为 $T_2$，故紧集 $M$ 到 $\phi[M]$ 的连续双射为同胚，且紧像 $\phi[M]$ 在 $N$ 中闭。
\end{proof}

\begin{theorem}[局部坐标面判据]\label{chdm:thm_embed-3}
    设 $\phi:M\to N$ 为单射浸入，$S:=\phi[M]$。则 $\phi$ 为嵌入，当且仅当每个 $q\in S$ 都有以 $q$ 为原点的坐标系 $(V,y)$，使
    \[
        S\cap V=\{r\in V:y^{m+1}(r)=\cdots=y^n(r)=0\}.
    \]
\end{theorem}
\begin{proof}
    若 $\phi$ 为嵌入，由浸入的局部标准形，取 $p=\phi^{-1}(q)$ 的开邻域 $U$，使 $\phi[U]$ 为局部坐标面。因 $\phi[U]$ 在 $S$ 中开，可取 $N$ 中开集 $W$ 使 $W\cap S=\phi[U]$。再将目标坐标域缩小到 $W$ 内的一个坐标开方体，便排除了 $U$ 外的点的像，得到所需等式。

    反之，给 $S$ 赋子拓扑。在每个上述坐标域内，$S\cap V$ 是 $m$ 维坐标面，且 $\phi:\phi^{-1}[V]\to S\cap V$ 为双射。因 $\phi$ 浸入，其作为两侧 $m$ 维流形间映射的切映射可逆，故为局部微分同胚。双射的局部微分同胚是同胚；这些坐标域覆盖 $S$，故 $\phi:M\to S$ 为同胚，即为嵌入。
\end{proof}

\begin{theorem}\label{chdm:thm_embed-2}
    浸入子流形 $S\subset N$ 是嵌入子流形，当且仅当它是 $N$ 的某个开子流形的闭子流形。
\end{theorem}
\begin{proof}
    若 $S$ 嵌入，取上一判据给出的坐标域 $\{V_\alpha\}$ 覆盖 $S$，则 $S\cap V_\alpha$ 在 $V_\alpha$ 中闭。令 $W:=\bigcup_\alpha V_\alpha$，有
    \[
        W\backslash S=\bigcup_\alpha(V_\alpha\backslash S),
    \]
    为开集，故 $S$ 为开子流形 $W$ 的闭子流形。反之，嵌入 $S\hookrightarrow W$ 与开集的包含映射 $W\hookrightarrow N$ 复合仍为嵌入，即得。
\end{proof}

对比定理 \ref{chdm:thm_immerse-1} 与 \ref{chdm:thm_embed-3}：浸入只要求一个源邻域的像是坐标面；嵌入则要求像集在目标邻域内的全部部分恰为这个坐标面。前者不排除远处的点再次进入目标邻域，后者还须控制这种整体的拓扑行为。例 \ref{chdm:exm_iec1} 展示自交，例 \ref{chdm:exm_iec2}、\ref{chdm:exm_iec3} 则说明即使没有自交，仍可能因远处像的积聚而不是嵌入。


\section{单参数变换群}\label{chdm:sec_One-Parameter-Transformations-Groups}

切矢场给出各点的瞬时运动方向与速度；沿它积分，便得到点随参数连续移动的变换。反之，对这样的变换求导又能恢复切矢场。这是单参数变换群与切矢场的联系。本节先用常微分方程建立这一对应，再说明非零切矢场可在局部化为坐标方向，以及 Lie 括号如何描述场沿流的变化。

以下流形与切矢场均光滑。我们使用常微分方程的局部存在唯一性及解对初值的光滑依赖；光滑系数已保证局部 Lipschitz 条件，无须另作假设。

\begin{definition}
    置 $v\in\mathfrak X(M)$。若开区间 $I\subset\R$ 上的光滑曲线 $\gamma:I\to M$ 满足
    \[
        \dot\gamma(t)=v_{\gamma(t)},
    \]
    则称 $\gamma$ 为 $v$ 的\textbf{积分曲线}，其像称为\textbf{轨道}。若 $v_p=0$，则称 $p$ 为 $v$ 的\textbf{奇点}，否则为\textbf{正则点}。
\end{definition}

由曲线切矢的坐标表达式 \eqref{chdm:eqn_DiffCurve-2}，在坐标域内记 $x(t):=(x^1\circ\gamma(t),\ldots,x^n\circ\gamma(t))$，积分曲线方程为
\eq{\label{chdm:eqn_tmp-va2xit}
    \frac{\d x^i(t)}{\d t}=v^i(x(t)),\qquad x^i(0)=x^i(p).
}
右端不显含 $t$，故平移参数仍为积分曲线；一般的重新参数化则会改变速度。奇点处的常值曲线 $\gamma(t)=p$ 是积分曲线，由唯一性，它也是以 $p$ 为初值的唯一解。

\begin{theorem}
    对每个 $p\in M$，存在唯一的\textbf{最大积分曲线} $\gamma_p:I_p\to M$，满足 $\gamma_p(0)=p$。其中 $I_p$ 为含 $0$ 的开区间；任一以 $p$ 为初值的积分曲线都是它的限制。
\end{theorem}
\begin{proof}
    局部存在唯一性给出 $0$ 附近的解。任意两个以 $p$ 为初值的解在共同区间内相等：相等的时刻组成闭集，又由局部唯一性为开集，故由区间连通性即得。于是可将所有这样的解拼接，定义域为各区间之并，仍是含 $0$ 的开区间。这便是唯一的最大解。
\end{proof}

\begin{example}
    $\R$ 上 $v=kx\p_x$ 的最大积分曲线为 $\gamma_p(t)=p\e^{kt}$，其中 $k\in\R$。在 $\R^2$ 上，$v=x\p_x+ky\p_y$ 的最大积分曲线为
    \[
        \gamma_{(a,b)}(t)=(a\e^t,b\e^{kt}),\qquad t\in\R.
    \]
    两式均由分离变量即得。在二维例中，$k=1$ 时非恒定轨道沿射线，$k=0$ 时沿水平线；$k<0$ 时两个坐标方向一伸一缩，原点附近呈鞍形。
\end{example}

\begin{proposition}\label{chdm:thm_st}
    若 $s\in I_p$ 且 $q=\gamma_p(s)$，则
    \[
        I_q=I_p-s,\qquad \gamma_q(t)=\gamma_p(s+t).
    \]
\end{proposition}
\begin{proof}
    $t\mapsto\gamma_p(s+t)$ 是以 $q$ 为初值的积分曲线，且仍为最大解，否则平移回去便延拓了 $\gamma_p$。由最大解的唯一性即得。
\end{proof}

因此，两条最大积分曲线的像一旦相交，便完全重合，只差参数平移。最大定义域始终是区间，不会在中间留有可补的空缺。

\begin{theorem}[流的存在性]\label{chdm:thm_vphil}
    置
    \eq{\label{chdm:eqn_tmp-gammap}
        \mathcal D:=\{(t,p):p\in M,\ t\in I_p\},\qquad
        \phi:\mathcal D\to M,\quad \phi(t,p):=\gamma_p(t).
    }
    则 $\mathcal D$ 是包含 $\{0\}\times M$ 的开集，$\phi$ 光滑，称为 $v$ 的\textbf{最大流}。记 $\phi_t(p):=\phi(t,p)$，则
    \[
        \phi_0(p)=p,\qquad \phi_t(\phi_s(p))=\phi_{t+s}(p),
    \]
    其中 $s\in I_p$，且 $t\in I_{\phi_s(p)}\iff s+t\in I_p$。对每个 $t$，$\phi_t$ 为开集 $M_t:=\{p:t\in I_p\}$ 到 $M_{-t}$ 的微分同胚，逆为 $\phi_{-t}$。
\end{theorem}
\begin{proof}
    解对初值的光滑依赖给出每个 $(0,p)$ 附近的光滑解映射。任给 $(t,p)\in\mathcal D$，沿 $\gamma_p$ 从 $0$ 到 $t$ 的紧参数段，可取有限个这样的局部解映射依次复合；在 $p$ 的一个邻域及 $t$ 附近仍有定义且光滑。由唯一性，它们与 $\phi$ 一致，故 $\mathcal D$ 开且 $\phi$ 光滑。

    复合律及其参数范围由命题 \ref{chdm:thm_st} 得到。特别地，对 $p\in M_t$，
    \[
        \phi_t(p)\in M_{-t},\qquad \phi_{-t}(\phi_t(p))=p.
    \]
    交换 $t,-t$ 同理，故 $\phi_t:M_t\to M_{-t}$ 为微分同胚。
\end{proof}

固定 $p$ 是沿一条积分曲线运动，固定 $t$ 则是将所有可运动到时刻 $t$ 的点同时移动。对于局部问题，只需取 $p$ 的开邻域 $U$ 及 $\epsilon>0$，使 $(-\epsilon,\epsilon)\times U\subset\mathcal D$。

\begin{definition}\label{chdm:def_1PTG-local}
    置开集 $U\subset M$。光滑映射 $\phi:(-\epsilon,\epsilon)\times U\to M$ 称为\textbf{局部单参数变换群}或\textbf{局部流}，若 $\phi_0(p)=p$，且
    \[
        \phi_t(\phi_s(p))=\phi_{t+s}(p)
    \]
    对所有满足 $p,\phi_s(p)\in U$ 及 $|s|,|t|,|s+t|<\epsilon$ 的参数成立。
\end{definition}

上述最大流在 $(-\epsilon,\epsilon)\times U$ 上的限制即为局部流。它的每个 $\phi_t:U\to\phi_t[U]$ 都是微分同胚，逆由最大流的 $\phi_{-t}$ 限制得到；不要求 $\phi_t[U]\subset U$。

\begin{definition}\label{chdm:def_vector-complete}
    若 $v$ 的每条最大积分曲线均定义在整个 $\R$ 上，即 $\mathcal D=\R\times M$，则称 $v$ \textbf{完备}。
    光滑映射 $\phi:\R\times M\to M$ 若满足 $\phi_0=\id$、$\phi_t\circ\phi_s=\phi_{t+s}$，则称为\textbf{单参数变换群}或\textbf{整体流}。
\end{definition}

此时 $\phi_t^{-1}=\phi_{-t}$，故各 $\phi_t$ 都是微分同胚，且 $\phi_t\circ\phi_s=\phi_s\circ\phi_t$。因此 $\{\phi_t:t\in\R\}$ 在复合下构成交换群，但不同参数可能给出同一变换。它也是加法群 $(\R,+)$ 在 $M$ 上的光滑作用；群作用的一般定义见 \S\ref{chlg:sec_Lie-transformation-group}。

\begin{theorem}\label{chdm:thm_1PTG}
    完备光滑切矢场与单参数变换群一一对应。由 $\phi$ 恢复的场称为其\textbf{无穷小生成元}，由
    \eq{\label{chdm:eqn_tmp-vecp}
        v_p(f)=\left.\frac{\d}{\d t}f(\phi_t(p))\right|_{t=0},\qquad
        f\in C^\infty_p(M)
    }
    给出。此式也适用于局部流，在其初值开集上恢复生成元。
\end{theorem}
\begin{proof}
    完备场的最大流满足上述群律。反之，式 \eqref{chdm:eqn_tmp-vecp} 是曲线 $t\mapsto\phi_t(p)$ 在 $0$ 处的切矢；其坐标分量为 $\partial_t\phi^i(0,p)$，故组成光滑场。对群律 $\phi_{t+h}(p)=\phi_h(\phi_t(p))$ 关于 $h$ 在 $0$ 处求导，得
    \[
        \frac{\d}{\d t}\phi_t(p)=v_{\phi_t(p)}.
    \]
    故各轨道均为 $v$ 的积分曲线，并定义在整个 $\R$ 上。由积分曲线的唯一性，两种构造互逆。局部流在轨道留于初值开集的时间内作同样计算即可。
\end{proof}

\begin{example}
    $\R$ 上的 $v=x^2\p_x$ 不完备。以 $1$ 为初值的最大积分曲线为
    \[
        \gamma_1(t)=\frac1{1-t},\qquad t\in(-\infty,1),
    \]
    在有限时间内趋于无穷，无法在 $\R$ 中继续延拓。另一方面，$\p_x$ 在 $\R$ 上完备，限制到 $(0,1)$ 后却不完备：$\gamma_p(t)=p+t$ 只在 $(-p,1-p)$ 内存在。可见完备性取决于场及其所在流形；仅缩短某条曲线的参数区间，不会改变场的完备性。
\end{example}

\begin{theorem}\label{chdm:thm_vphig}
    紧流形上的光滑切矢场均完备。
\end{theorem}
\begin{proof}
    由局部存在性，每点有初值邻域 $U$ 及共同存在时间 $\epsilon_U>0$。取有限个这样的 $U$ 覆盖 $M$，再取各 $\epsilon_U$ 的最小值 $\epsilon>0$，则任意初值的解至少存在于 $(-\epsilon,\epsilon)$。若某条最大积分曲线的右端点 $b$ 有限，取 $t_0<b$ 使 $b-t_0<\epsilon$，从 $\gamma(t_0)$ 出发的解便能将它延拓到 $b$ 以后，矛盾。左端点同理。
\end{proof}

下面考察流在微分同胚下如何变换。推前的定义沿用前文，故只需对轨道求导。
\begin{theorem}\label{chdm:thm_1PDG-diff}
    设 $\phi_t$ 是 $v$ 生成的单参数变换群，$\psi:M\to M$ 为微分同胚。则
    \[
        \tilde\phi_t:=\psi\circ\phi_t\circ\psi^{-1}
    \]
    是 $\psi_*v$ 生成的单参数变换群。
\end{theorem}
\begin{proof}
    $\tilde\phi_0=\id$，且 $\tilde\phi_t\circ\tilde\phi_s=\psi\circ\phi_{t+s}\circ\psi^{-1}=\tilde\phi_{t+s}$。置 $q=\psi(p)$，由链式法则，
    \[
        \left.\frac{\d}{\d t}\tilde\phi_t(q)\right|_{t=0}
        =\psi_{*p}\left(\left.\frac{\d}{\d t}\phi_t(p)\right|_{t=0}\right)
        =\psi_{*p}v_p=(\psi_*v)_q.
    \]
\end{proof}

\begin{definition}\label{chdm:def_invariant-vector-field}
    若微分同胚 $\psi:M\to M$ 满足 $\psi_*v=v$，则称 $v$ 在 $\psi$ 下\textbf{不变}。
\end{definition}

\begin{proposition}\label{chdm:thm_ivf}
    设 $\phi_t$ 为完备场 $v$ 的流。则 $\psi_*v=v$ 当且仅当 $\psi\circ\phi_t=\phi_t\circ\psi$ 对所有 $t$ 成立。
\end{proposition}
\begin{proof}
    由上一定理，$\psi\circ\phi_t\circ\psi^{-1}$ 的生成元为 $\psi_*v$。它与 $\phi_t$ 相同，当且仅当生成元相同，故结论成立。
\end{proof}

特别地，流保持自身的生成元：
\eq{\label{chdm:eqn_phisv}
    (\phi_s)_{*p}v_p=v_{\phi_s(p)}.
}
此式对不完备场的最大流也成立：将 $\phi_s(\gamma_p(t))=\gamma_p(s+t)$ 在 $t=0$ 处求导即可。

前文已定义 Lie 括号；现在可用流解释它。不同点的切矢不能直接相减，须先用流的切映射将它们送到同一切空间。
\begin{theorem}\label{chdm:thm_xy-yx}
    设 $X,Y\in\mathfrak X(M)$，$\phi$ 为 $X$ 的最大流。对任意 $p\in M$，
    \eq{\label{chdm:eqn_xy-yx}
    \begin{aligned}
        [X,Y]_p
        &=\lim_{t\to0}\frac{(\phi_{-t})_{*\phi_t(p)}Y_{\phi_t(p)}-Y_p}{t}\\
        &=\lim_{t\to0}\frac{Y_p-(\phi_t)_{*\phi_{-t}(p)}Y_{\phi_{-t}(p)}}{t}.
    \end{aligned}
    }
    两个极限均在 $T_pM$ 中求取，故不要求 $X$ 完备。
\end{theorem}
\begin{proof}
    两式由 $t\mapsto-t$ 互换，只证第一式。取 $p$ 附近的光滑函数 $f$，置 $H(t,q):=Y_q(f\circ\phi_{-t})$。则
    \[
        \bigl((\phi_{-t})_{*\phi_t(p)}Y_{\phi_t(p)}\bigr)(f)
        =H(t,\phi_t(p)).
    \]
    因 $H(0,q)=Yf(q)$，且 $\partial_t(f\circ\phi_{-t})|_{t=0}=-Xf$，由链式法则及光滑偏导的交换性，
    \[
        \left.\frac{\d}{\d t}H(t,\phi_t(p))\right|_{t=0}
        =-Y_p(Xf)+X_p(Yf)=[X,Y]_p(f).
    \]
    对各坐标函数取 $f=x^i$ 即得各分量的极限，故切矢极限成立。
\end{proof}

最后说明非奇点附近的流为何具有简单形状：取一片与场横截的坐标面，以面上坐标标记各条轨道，再以运动时间作为剩下的坐标。
\begin{theorem}[矢量场直化定理]\label{chdm:thm_1PDG-ppy}
    若 $X_p\ne0$，则存在 $p$ 附近的坐标系 $(V;y^1,\ldots,y^n)$，使 $X|_V=\partial/\partial y^1$。这称为 $X$ 的\textbf{适配坐标系}。
\end{theorem}
\begin{proof}
    取以 $p$ 为原点的坐标系 $x$，不妨设 $X^1(p)\ne0$。记 $\phi$ 为 $X$ 的最大流，在原点附近定义
    \[
        F(t,a^2,\ldots,a^n):=\phi_t\bigl(x^{-1}(0,a^2,\ldots,a^n)\bigr).
    \]
    在原点，$F_*\partial_t=X_p$，$F_*\partial_{a^\alpha}=\partial_{x^\alpha}|_p$（$\alpha\geqslant2$）；由 $X^1(p)\ne0$，这些切矢线性无关。逆映射定理说明 $F$ 在缩小的邻域上为微分同胚。取 $y:=F^{-1}$ 为坐标，因 $F_*\partial_t=X$，便有 $X=\partial/\partial y^1$。
\end{proof}

在适配坐标中，流只是平移 $y^1$：$\phi_t(y)=(y^1+t,y^2,\ldots,y^n)$，只要这段轨道仍在坐标域内。奇点处不能如此直化，因为坐标基矢处处非零。


\section{Lie 导数}\label{appx:Lie-deri}

前节已由切矢场 $X$ 构造出流 $\phi_t$。沿流观察张量场 $T$ 时，$T_p$ 与 $T_{\phi_t(p)}$ 属于不同点的张量空间，不能直接相减。流的拉回恰好将后者送回 $p$，于是可以比较 $(\phi_t^*T)_p$ 与 $T_p$。它们的一阶差异就是 Lie 导数。对切矢，拉回由逆流的推前实现；对余切矢，则直接使用流的拉回。这些运算已在“诱导映射”中定义，无须另选度规或联络。

\begin{definition}
    置 $X\in\mathfrak X(M)$，记其最大流为 $\phi$。光滑 $(k,l)$-张量场 $T$ 沿 $X$ 的 \textbf{Lie 导数}\footnote{Sophus Lie（1842--1899），挪威数学家。李群见第 \ref{appx:Lie-group} 章。}定义为
    \eq{\label{chdm:def_LieD-tensor}
        (\Lie_XT)_p:=\left.\frac{\d}{\d t}(\phi_t^*T)_p\right|_{t=0}
        =\lim_{t\to0}\frac{(\phi_t^*T)_p-T_p}{t}.
    }
    极限在同一张量空间 $\mathcal T^k_l(p)$ 中求取。流及其切映射光滑，故该极限存在，并组成同型的光滑张量场。定义只用 $t=0$ 附近的流，不要求 $X$ 完备。
\end{definition}

对标量场，拉回只是复合，故
\eq{\label{chdm:def_LieD-scalar}
    \Lie_Xf=\left.\frac{\d}{\d t}(f\circ\phi_t)\right|_{t=0}=X(f).
}
对切矢场，$\phi_t^*Y=(\phi_{-t})_*Y$，故
\eq{\label{chdm:def_LieD-vector}
    \Lie_XY=\lim_{t\to0}\frac{(\phi_{-t})_*Y-Y}{t}.
}
由前节定理 \ref{chdm:thm_xy-yx}，这正是 Lie 括号：
\eq{\label{chdm:eqn_LieD-tangent}
    \Lie_XY=[X,Y].
}
因此 Lie 导数把函数的方向导数、切矢场的 Lie 括号统一推广到了任意张量场。

\begin{theorem}\label{chdm:thm_LieD-property}
    Lie 导数保持张量类型，并满足实线性、Leibniz 法则及与缩并的交换性：
    \[
    \begin{aligned}
        \Lie_X(aT+bS)&=a\Lie_XT+b\Lie_XS,\qquad a,b\in\R,\\
        \Lie_X(T\otimes S)&=(\Lie_XT)\otimes S+T\otimes\Lie_XS,\\
        \Lie_X(CT)&=C(\Lie_XT).
    \end{aligned}
    \]
    其中加法要求 $T,S$ 同型，$C$ 表示任一缩并。若 $\psi:M\to M$ 为微分同胚，还有
    \[
        \psi_*(\Lie_XT)=\Lie_{\psi_*X}(\psi_*T).
    \]
\end{theorem}
\begin{proof}
    前文命题 \ref{chdm:thm_pptt} 已证明拉回保持线性、张量积及缩并。将对应等式对流参数在 $t=0$ 处求导，前三式即得。特别地，张量积的求导就是乘积法则。

    定理 \ref{chdm:thm_1PDG-diff} 的共轭流证明同样适用于局部流，故 $\psi_*X$ 的局部流为 $\tilde\phi_t=\psi\circ\phi_t\circ\psi^{-1}$。利用 $\psi_*=(\psi^{-1})^*$ 及拉回的复合律，得
    \[
        \psi_*\circ\phi_t^*=\tilde\phi_t^*\circ\psi_*.
    \]
    两边在 $t=0$ 处求导，即得最后一式。
\end{proof}

\begin{proposition}
    对余切矢场 $\omega$ 及任意切矢场 $Y$，
    \eq{\label{chdm:eqn_LieD-cotangent}
        (\Lie_X\omega)(Y)=X\bigl(\omega(Y)\bigr)-\omega([X,Y]).
    }
    特别地，对任意光滑函数 $f$，$\Lie_X(\d f)=\d(Xf)$。
\end{proposition}
\begin{proof}
    对配对 $\omega(Y)$ 应用 Leibniz 法则及缩并交换性，得
    \[
        X\bigl(\omega(Y)\bigr)=(\Lie_X\omega)(Y)+\omega(\Lie_XY),
    \]
    代入 $\Lie_XY=[X,Y]$ 即得第一式。再取 $\omega=\d f$，其右端为
    \[
        X(Yf)-[X,Y]f=Y(Xf)=\d(Xf)(Y),
    \]
    故第二式成立。
\end{proof}

以上运算规则给出了简便的坐标算法。在坐标系 $\{x^i\}$ 中，
\eq{\label{chdm:eqn_LieD-bases}
    \Lie_X\p_i=[X,\p_i]=-(\p_iX^a)\p_a,\qquad
    \Lie_X(\d x^j)=\d X^j=(\p_aX^j)\d x^a.
}
将张量场展开为分量与坐标基的张量积，再逐项使用 Leibniz 法则，就能求出一般表达式。

\begin{theorem}[Lie 导数的分量]
    对任意 $(k,l)$-张量场 $T$，
    \eq{\label{chdm:eqn_LieD-components}
    \begin{aligned}
        (\Lie_XT)^{i_1\cdots i_k}{}_{j_1\cdots j_l}
        ={}&X^a\p_aT^{i_1\cdots i_k}{}_{j_1\cdots j_l}\\
        &-\sum_{r=1}^k(\p_aX^{i_r})
        T^{i_1\cdots a\cdots i_k}{}_{j_1\cdots j_l}\\
        &+\sum_{s=1}^l(\p_{j_s}X^a)
        T^{i_1\cdots i_k}{}_{j_1\cdots a\cdots j_l}.
    \end{aligned}
    }
    两个求和项分别将第 $r$ 个上指标、第 $s$ 个下指标换为 $a$。
\end{theorem}
\begin{proof}
    对分量函数求导给出首项。由式 \eqref{chdm:eqn_LieD-bases}，每个切基因子给出一个负号项，每个余切基因子给出一个正号项，合并即得。
\end{proof}

首项只是对分量作方向导数；其余各项来自基底随流的变换，不能在一般坐标系中省略。例如，
\[
\begin{aligned}
    (\Lie_XY)^i&=X^a\p_aY^i-Y^a\p_aX^i,\\
    (\Lie_X\omega)_i&=X^a\p_a\omega_i+\omega_a\p_iX^a,\\
    (\Lie_XT)^{ij}&=X^a\p_aT^{ij}-T^{aj}\p_aX^i-T^{ia}\p_aX^j,\\
    (\Lie_XS)_{ij}&=X^a\p_aS_{ij}+S_{aj}\p_iX^a+S_{ia}\p_jX^a.
\end{aligned}
\]

\begin{corollary}
    Lie 导数对生成场也具有实线性：
    \[
        \Lie_{aX+bY}T=a\Lie_XT+b\Lie_YT,\qquad a,b\in\R.
    \]
    但一般不具有函数线性。
\end{corollary}
\begin{proof}
    分量公式对 $X^i$ 及其一阶偏导均线性，故第一式成立。若将常数换成函数，求导会产生额外项；例如
    \[
        \Lie_{fX}Y=[fX,Y]=f\Lie_XY-Y(f)X.
    \]
\end{proof}

\begin{example}[适配坐标]
    若 $X_p\ne0$，由直化定理可在 $p$ 附近取坐标使 $X=\p_1$。此时 $X$ 的分量恒定，上述基底变化项全为零，故对任意型张量场，
    \eq{\label{chdm:eqn_LieD-adaptedX}
        (\Lie_XT)^{i_1\cdots i_k}{}_{j_1\cdots j_l}
        =\p_1T^{i_1\cdots i_k}{}_{j_1\cdots j_l}
        =X\bigl(T^{i_1\cdots i_k}{}_{j_1\cdots j_l}\bigr).
    }
    几何上，流只平移第一个坐标，其 Jacobi 矩阵为单位矩阵，故拉回保持坐标基及对偶基，只需比较分量函数。这里张量 Lie 导数的分量恰等于分量的方向导数，是适配坐标的特殊性质。
\end{example}

\begin{example}
    在 $\R^2$ 上取 $X=x^2\p_x$、$Y=\p_x+y^2\p_y$。由 Lie 括号的分量式，
    \[
        \Lie_XY=[X,Y]=-2x\p_x.
    \]
    换用 $u=x+y$、$v=x-y$，有 $x=(u+v)/2$、$\p_x=\p_u+\p_v$，故
    \[
        \Lie_XY=-(u+v)(\p_u+\p_v).
    \]
    这是同一切矢场在两套坐标中的表达。
\end{example}

Lie 导数还可用来刻画流所保持的张量场。
\begin{theorem}
    $\Lie_XT=0$ 当且仅当 $X$ 的流保持 $T$，即 $\phi_t^*T=T$ 在各自的定义域上成立。
\end{theorem}
\begin{proof}
    由流的复合律及拉回的复合律，在任一允许的时刻 $t$ 有
    \[
        \frac{\d}{\d t}\phi_t^*T
        =\left.\frac{\d}{\d h}\phi_t^*(\phi_h^*T)\right|_{h=0}
        =\phi_t^*(\Lie_XT).
    \]
    若 $\Lie_XT=0$，则对每个 $p$，$(\phi_t^*T)_p$ 在区间 $I_p$ 上恒为 $T_p$。反之，将 $\phi_t^*T=T$ 在 $t=0$ 处求导即得。
\end{proof}

特别地，$\Lie_Xg=0$ 表示流保持度规，$\Lie_XY=[X,Y]=0$ 表示流保持切矢场 $Y$。在时空中，光滑单位类时场 $Z$ 的积分曲线描述观察者的世界线；沿这些世界线考察 $\Lie_ZT$，便是在用观察者线汇的流比较邻近的张量。若它为零，$T$ 随该流保持不变，此性质与坐标选择无关。

\section{Killing 矢量}
定义流形间映射便可谈及微分同胚。微分同胚映射可自然诱导张量丛的映射，故可谈及张量变换的主动、被动观点。所有微分同胚构成微分同胚群。任意光滑矢量场是该群的生成元。显然微分同胚有无穷多生成元，故维数无穷。一个光滑矢量场在群中挑出一个单参子群，称为单参微分同胚局部群（无穷小微分同胚）。若光滑矢量场还有完备性，则给出一个单参微分同胚群。紧致流形上任意矢量场可延拓至完备的。保持度规分量不变的变换对应的微分同胚映射称为等度规映射。诱导此映射的矢量场称为 Killing 矢量。代数定义可从 Lie 导数出发，也可从 Killing 方程出发以跳过 Lie 导数。此定义便告知可用 Killing 矢量场讨论度规（时空）的对称性。
